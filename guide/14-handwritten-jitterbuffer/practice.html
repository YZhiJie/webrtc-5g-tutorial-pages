14.3 实战案例：Node.js 手写 JitterBuffer 模拟器 ​

只用 Node.js 内置模块，从零实现一个带乱序重排、丢包判定、抖动估计的 JitterBuffer，再配一个网络模拟器和播放器，在终端里完整复现"抖动 → 缓冲 → 平滑播放"的全过程。

14.3.1 我们要模拟什么 ​

发送端                     网络                      接收端
┌────────┐   每 20ms 一包   ┌──────────────┐   ┌─────────────────┐   ┌────────┐
│ 帧产生器 │ ──────────────► │ 延迟/乱序/丢包 │ ─► │  JitterBuffer   │ ─► │ 播放器 │
└────────┘                 └──────────────┘   │  - 乱序重排      │   │ 每20ms │
   seq=0..99                  5% 丢包          │  - 丢包等待/放弃  │   │ 取一帧 │
                              20% 额外抖动      │  - 抖动估计      │   └────────┘
                                              └─────────────────┘

三个文件：

文件职责
jitter-buffer.jsJitterBuffer 类：push(packet) 收包，getFrame(targetTime) 取帧
network-sim.js网络模拟器：加延迟、乱序、丢包
player.js播放器：每 20ms 一个 tick，调用 getFrame(now - bufferDepth) 并统计

14.3.2 第一步：JitterBuffer 核心（jitter-buffer.js） ​

js
'use strict';

/**
 * JitterBuffer：接收端的抖动缓冲区
 * - push(packet)：收到 RTP 包时调用（包可能乱序、重复、迟到）
 * - getFrame(targetTime)：播放器每 tick 调用，取出到期的完整帧
 *
 * packet = { seq, timestamp, data, marker, arrivalTime }
 *   seq         RTP 序列号（Sequence Number），每发一包 +1
 *   timestamp   发送端打的时间戳（本例即该帧的播放时刻，单位 ms）
 *   marker      帧结束标记（M bit），本例一包一帧，恒为 true
 *   arrivalTime 到达接收端的本地时刻（由网络模拟器填写）
 */
class JitterBuffer {
  constructor({ minDepthMs = 60, maxDepthMs = 200, maxWaitMs = 100 } = {}) {
    this.minDepthMs = minDepthMs; // 缓冲下限：再低就容易"断粮"
    this.maxDepthMs = maxDepthMs; // 缓冲上限：再高端到端延迟就太大
    this.maxWaitMs = maxWaitMs;   // kMaxWaitMs：等一个缺失的包最多多久

    this.packets = new Map();     // seq -> packet，乱序重排的核心结构
    this.nextSeq = null;          // 下一个期望播放的 seq
    this.gapSince = null;         // 第一次发现 nextSeq 空洞时的 targetTime

    // ---- 抖动统计（RFC 3550 附录 A.8 的经典算法）----
    this.jitter = 0;              // 抖动估计值 J
    this.prevTransit = null;      // 上一个包的 transit time
  }

  push(packet) {
    if (this.packets.has(packet.seq)) return;                    // 重复包，丢弃
    if (this.nextSeq !== null && packet.seq < this.nextSeq) return; // 来得太晚，已被跳过
    this.packets.set(packet.seq, packet);
    if (this.nextSeq === null || packet.seq < this.nextSeq) {
      this.nextSeq = packet.seq;                                 // 乱序到达时向前修正起点
    }
    // transit time = 到达时刻 - 发送时刻；相邻包的差值 |D| 即瞬时抖动
    const transit = packet.arrivalTime - packet.timestamp;
    if (this.prevTransit !== null) {
      const d = Math.abs(transit - this.prevTransit);
      this.jitter += (d - this.jitter) / 16;                     // 指数滑动平均
    }
    this.prevTransit = transit;
  }

  /**
   * 取一帧。返回 { status, frame? }
   *   'ok'      正常取到帧
   *   'waiting' 帧还没到播放时间，或包还在路上（乱序）
   *   'missing' 等了 maxWaitMs 仍没到齐，判定丢失并跳过
   */
  getFrame(targetTime) {
    if (this.nextSeq === null) return { status: 'waiting' };

    const pkt = this.packets.get(this.nextSeq);
    if (pkt) {
      if (pkt.timestamp <= targetTime) {
        this.packets.delete(this.nextSeq);
        this.nextSeq++;
        this.gapSince = null;
        return { status: 'ok', frame: pkt };
      }
      return { status: 'waiting' }; // 包已到但还没到期，继续囤着
    }

    // nextSeq 缺失：可能还在路上（乱序），也可能真丢了
    if (this.gapSince === null) {
      this.gapSince = targetTime; // 记录第一次发现空洞的时刻
      return { status: 'waiting' };
    }
    if (targetTime - this.gapSince >= this.maxWaitMs) {
      const seq = this.nextSeq;
      this.nextSeq++;             // 放弃等待，跳过这个 seq
      this.gapSince = null;
      return { status: 'missing', seq };
    }
    return { status: 'waiting' };
  }

  /** 当前缓冲里囤了多少毫秒的帧（自适应调深度的输入之一） */
  bufferedMs() {
    if (this.packets.size === 0) return 0;
    let min = Infinity, max = -Infinity;
    for (const p of this.packets.values()) {
      if (p.timestamp < min) min = p.timestamp;
      if (p.timestamp > max) max = p.timestamp;
    }
    return max - min;
  }

  /** 根据抖动估计给出的建议缓冲深度，夹在 [minDepthMs, maxDepthMs] 之间 */
  suggestDepthMs() {
    const d = this.minDepthMs + this.jitter * 4;
    return Math.round(Math.min(this.maxDepthMs, Math.max(this.minDepthMs, d)));
  }
}

module.exports = { JitterBuffer };

14.3.3 第二步：网络模拟器（network-sim.js） ​

js
'use strict';

/** 可复现的伪随机数发生器（mulberry32），固定 seed 让每次运行输出一致 */
function mulberry32(seed) {
  let a = seed >>> 0;
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/**
 * 网络模拟器：给包加延迟 / 乱序 / 丢包
 * - 每个包有 1~10ms 的基础延迟
 * - reorderRate 比例的包再额外延迟 50~200ms（制造乱序与抖动）
 * - lossRate 比例的包直接丢弃
 *
 * 用法：发送端调 send(packet, sendTime)；接收端每 tick 调 advanceTo(now)，
 * 到达的包会通过 onDeliver 回调交给 JitterBuffer。
 */
function createNetworkSim({ lossRate = 0.05, reorderRate = 0.2, onDeliver, seed = 42 }) {
  const rand = mulberry32(seed);
  const pending = []; // { deliverAt, packet }，简易事件队列

  return {
    send(packet, sendTime) {
      if (rand() < lossRate) return;                          // 丢包：根本不上路
      let delay = 1 + rand() * 9;                             // 基础延迟 1~10ms
      if (rand() < reorderRate) delay += 50 + rand() * 150;   // 抖动包：再 +50~200ms
      pending.push({ deliverAt: sendTime + delay, packet });
    },

    /** 时钟推进到 now，把所有到达的包投递给接收端 */
    advanceTo(now) {
      pending.sort((a, b) => a.deliverAt - b.deliverAt);
      while (pending.length && pending[0].deliverAt <= now) {
        const { deliverAt, packet } = pending.shift();
        packet.arrivalTime = Math.round(deliverAt);
        onDeliver(packet);
      }
    },
  };
}

module.exports = { createNetworkSim };

为什么用虚拟时钟而不是 setInterval？ 虚拟时钟让整个模拟瞬间跑完、结果可复现（固定 seed），不依赖真实耗时，适合做单元测试。

14.3.4 第三步：播放器（player.js） ​

js
'use strict';

const { JitterBuffer } = require('./jitter-buffer');
const { createNetworkSim } = require('./network-sim');

const TOTAL_FRAMES = 100;    // 发送 100 帧
const TICK_MS = 20;          // 每 20ms 一帧；播放器也每 20ms 取一帧
const BUFFER_DEPTH_MS = 120; // 播放延迟：收到的帧至少囤 120ms 再播

const jb = new JitterBuffer({ minDepthMs: 60, maxDepthMs: 200, maxWaitMs: 100 });

// ---- 发送端：每 20ms 产生一包（一包 = 一帧），交给网络模拟器 ----
const net = createNetworkSim({
  lossRate: 0.05,   // 5% 丢包
  reorderRate: 0.2, // 20% 的包被额外延迟 50~200ms
  seed: 42,
  onDeliver: (pkt) => jb.push(pkt),
});
for (let i = 0; i < TOTAL_FRAMES; i++) {
  net.send(
    { seq: i, timestamp: i * TICK_MS, data: `frame-${i}`, marker: true },
    i * TICK_MS
  );
}

// ---- 播放端：虚拟时钟每 20ms 走一格 ----
const stats = { ok: 0, missing: 0, buffering: 0, stallTicks: 0, stallEvents: 0 };
let started = false;  // 第一次取到帧后视为起播
let stalling = false; // 当前是否处于停顿中
const SIM_END = TOTAL_FRAMES * TICK_MS + 2000;

console.log(`开始模拟：${TOTAL_FRAMES} 帧，tick=${TICK_MS}ms，播放延迟=${BUFFER_DEPTH_MS}ms\n`);

for (let now = 0; now <= SIM_END; now += TICK_MS) {
  net.advanceTo(now);                      // 1. 把到达的包收进缓冲
  if (jb.nextSeq >= TOTAL_FRAMES) break;   //    所有帧都有结论了，收尾

  const target = now - BUFFER_DEPTH_MS;    // 2. 计算本次的播放目标时刻
  const r = jb.getFrame(target);           // 3. 向 JitterBuffer 要一帧

  const tag = r.status === 'ok' ? '正常' : r.status === 'missing' ? '缺失' : '等待';
  const extra = r.status === 'ok' ? ` 播放 seq=${r.frame.seq}`
              : r.status === 'missing' ? ` 放弃 seq=${r.seq}` : '';
  console.log(`t=${String(now).padStart(4)}ms  target=${String(target).padStart(4)}ms  ${tag}${extra}`);

  if (r.status === 'ok') {
    stats.ok++; started = true; stalling = false;
  } else if (r.status === 'missing') {
    stats.missing++; stalling = false;
  } else if (!started) {
    stats.buffering++;                     // 起播前的等待算"缓冲中"
  } else {
    stats.stallTicks++;                    // 起播后的等待 = 卡顿
    if (!stalling) { stats.stallEvents++; stalling = true; }
  }
}

console.log('\n================ 统计 ================');
console.log(`总帧数:            ${TOTAL_FRAMES}`);
console.log(`正常播放:          ${stats.ok}`);
console.log(`丢失帧数:          ${stats.missing}`);
console.log(`起播缓冲 tick:     ${stats.buffering}`);
console.log(`停顿 tick 数:      ${stats.stallTicks}（约 ${stats.stallTicks * TICK_MS}ms）`);
console.log(`停顿次数:          ${stats.stallEvents}`);
console.log(`抖动估计 J:        ${jb.jitter.toFixed(1)}ms`);
console.log(`建议缓冲深度:      ${jb.suggestDepthMs()}ms（当前固定 ${BUFFER_DEPTH_MS}ms）`);

14.3.7 与真实 WebRTC 的差距 ​

本模拟器做了大量简化，真实 WebRTC 还要处理：

简化点真实实现
一包一帧视频帧分多包，需按 timestamp + M bit 组装（见 14.2.1）
seq 不回绕16 位 seq 需处理 wrap-around（见 14.2.2）
固定播放延迟NetEq / JitterBuffer 用 IAT 直方图自适应调整（见 14.2.5）
丢包直接放弃先发 NACK 请求重传，等 kMaxWaitMs 仍没到才放弃（见 14.2.3）
音频不做变速NetEq 会加速/减速播放来消化积压或争取时间（见 14.2.4）
