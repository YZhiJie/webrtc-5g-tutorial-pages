16.3 实战案例：Node.js 模拟 DTLS 握手与 SRTP 加解密 ​

本章节用纯 Node.js 内置模块（crypto + dgram）实现 WebRTC 媒体加密的最小可用链路：ECDH 握手 → 证书指纹验证 → 密钥导出 → SRTP KDF → AES-CM 加密 → HMAC-SHA1 认证 → UDP 传输 → 解密还原。你可以直接复制代码到本地运行，亲眼看到明文 "hello world" 变成密文、经过 UDP 传输后被正确解密回来的全过程。

16.3.1 项目结构 ​

创建三个文件在同一目录下：

srtp-demo/
├── dtls-lite.js   # 简化 DTLS 握手：ECDH 密钥交换 + 指纹验证 + 密钥导出
├── srtp.js        # SRTP：KDF 密钥派生 + AES-CM 加密 + HMAC-SHA1 认证
└── main.js        # 主流程：构造 RTP 包 → 加密 → UDP 发送 → 解密验证

整体数据流：

                 dtls-lite.js                        srtp.js
            ┌───────────────────┐         ┌──────────────────────┐
 Client ─── │ ECDH 密钥交换      │ ──────> │ KDF 派生会话密钥       │
            │ 证书指纹验证       │         │ AES-CM 加密 payload   │
 Server ─── │ Exporter 导出密钥  │ ──────> │ HMAC-SHA1 认证标签    │
            └───────────────────┘         └──────────┬───────────┘
                                                     │
                                              UDP (dgram) 传输
                                                     │
                                            Server unprotect
                                            验标签 → 解密 → 明文

16.3.2 dtls-lite.js：简化 DTLS 握手 ​

真实 DTLS 有 6~7 条 UDP 消息往返和完整状态机，这里只保留密码学核心：ECDH 密钥交换、证书指纹验证、SRTP 密钥材料导出。

js
// dtls-lite.js — 简化 DTLS 握手模拟
// 只保留核心密码学步骤：ECDH 密钥交换 + 证书指纹验证 + SRTP 密钥材料导出
// 只用 Node.js 内置模块 crypto

const crypto = require('crypto')

// RFC 5764 规定的固定导出标签
const EXPORTER_LABEL = 'EXTRACTOR-dtls_srtp'

// ---------- 生成 ECDH 密钥对（用于握手密钥交换，前向保密） ----------
function generateEcdhKeyPair() {
  const ecdh = crypto.createECDH('prime256v1')
  ecdh.generateKeys()
  return ecdh
}

// ---------- 生成"自签名证书" ----------
// 真实 DTLS 用 X.509 证书；这里简化为：RSA 公钥的 DER 编码 + 其 SHA-256 指纹
// 指纹通过信令通道（假设安全）传递给对端，对应 SDP 中的 a=fingerprint 行
function generateSelfSignedCert() {
  const { publicKey, privateKey } = crypto.generateKeyPairSync('rsa', {
    modulusLength: 2048,
  })
  // 用 SPKI DER 编码模拟证书 DER
  const certDER = publicKey.export({ type: 'spki', format: 'der' })
  const fingerprint = crypto.createHash('sha256')
    .update(certDER)
    .digest('hex')
    .toUpperCase()
    .match(/.{2}/g)
    .join(':')
  return { publicKey, privateKey, certDER, fingerprint }
}

// ---------- TLS Exporter（RFC 5705）简化版 ----------
// 真实实现基于 TLS PRF；这里用内置 HKDF 模拟"从握手密钥导出独立密钥材料"
function tlsExporter(sharedSecret, label, length) {
  const out = crypto.hkdfSync('sha256', sharedSecret, Buffer.alloc(0), Buffer.from(label), length)
  return Buffer.from(out)
}

// ---------- 按 RFC 5764 切分 60 字节密钥材料 ----------
// [0:16] client_master_key | [16:30] client_master_salt
// [30:46] server_master_key | [46:60] server_master_salt
function splitKeyingMaterial(km) {
  return {
    clientMasterKey: km.subarray(0, 16),
    clientMasterSalt: km.subarray(16, 30),
    serverMasterKey: km.subarray(30, 46),
    serverMasterSalt: km.subarray(46, 60),
  }
}

// ---------- 模拟完整握手 ----------
// 真实场景中以下步骤是 6~7 条 UDP 消息往返；这里同步函数模拟双方状态推进
function dtlsHandshake() {
  console.log('--- DTLS 握手开始 ---')

  // 1. ClientHello：客户端生成临时 ECDH 密钥对，发送公钥
  const clientEcdh = generateEcdhKeyPair()
  console.log('[Client] 发送 ClientHello（含 ECDH 公钥、use_srtp 扩展）')

  // 2. ServerHello + Certificate：服务器生成 ECDH 密钥对与自签名证书
  const serverEcdh = generateEcdhKeyPair()
  const serverCert = generateSelfSignedCert()
  console.log('[Server] 发送 ServerHello + Certificate')
  console.log('[Server] 证书指纹（通过信令通道预先告知客户端）:')
  console.log('         a=fingerprint:sha-256 ' + serverCert.fingerprint)

  // 3. 客户端验证证书指纹（与信令通道收到的指纹对比）
  const computedFingerprint = crypto.createHash('sha256')
    .update(serverCert.certDER)
    .digest('hex')
    .toUpperCase()
    .match(/.{2}/g)
    .join(':')
  if (computedFingerprint !== serverCert.fingerprint) {
    throw new Error('证书指纹不匹配！可能遭遇 MITM 攻击')
  }
  console.log('[Client] ✓ 证书指纹验证通过')

  // 4. 双方各自计算 ECDH 共享密钥（pre-master secret 的角色）
  const clientSecret = clientEcdh.computeSecret(serverEcdh.getPublicKey())
  const serverSecret = serverEcdh.computeSecret(clientEcdh.getPublicKey())
  if (!clientSecret.equals(serverSecret)) {
    throw new Error('共享密钥不一致！')
  }
  console.log('[Both]   ✓ ECDH 共享密钥:', clientSecret.toString('hex'))

  // 5. 通过 Exporter 导出 SRTP 密钥材料（60 字节）
  const km = tlsExporter(clientSecret, EXPORTER_LABEL, 60)
  console.log('[Both]   ✓ Exporter("' + EXPORTER_LABEL + '") 导出 60 字节:')
  console.log('         ', km.toString('hex'))

  console.log('--- DTLS 握手完成，ChangeCipherSpec/Finished 双方验证通过 ---\n')
  return splitKeyingMaterial(km)
}

module.exports = { dtlsHandshake, tlsExporter, EXPORTER_LABEL }

与真实 DTLS 的差异：真实握手还包括 cookie 防伪造（HelloVerifyRequest）、消息重传定时器、Finished 消息对全部握手内容做 MAC。这里全部省略，只保留密钥建立的最短路径。

16.3.3 srtp.js：KDF、AES-CM 加密与 HMAC-SHA1 认证 ​

这是本模块的核心：按 RFC 3711 实现密钥派生和逐包加解密。

js
// srtp.js — 简化 SRTP：AES-128-CM 加密 + HMAC-SHA1 认证（80-bit 截断）
// 只用 Node.js 内置模块 crypto

const crypto = require('crypto')

const RTP_HEADER_LEN = 12 // 简化：固定 12 字节头（无 CSRC/扩展）
const AUTH_TAG_LEN = 10   // SRTP_AES128_CM_HMAC_SHA1_80 → 80 bit = 10 字节

// ========== RFC 3711 §4.3 密钥派生函数（KDF），kdr = 0 ==========
// 输入 master key (16B) + master salt (14B)，按 label 派生不同用途的会话密钥
function srtpKdf(masterKey, masterSalt, label, length) {
  // x = key_id XOR master_salt
  // key_id = <label:1字节> || r:6字节（kdr=0 时 r=0），与 14 字节 salt 右对齐异或
  // → label 落在 salt 的第 7 字节（libsrtp 同此实现）
  const x = Buffer.alloc(16)
  masterSalt.copy(x, 0, 0, 14) // 14 字节盐放高位，末尾 2 字节为 0（等价于 x * 2^16）
  x[7] ^= label

  // AES-CM PRF：keystream_block(i) = AES-ECB(masterKey, x*2^16 + i)
  let out = Buffer.alloc(0)
  let counter = 0
  while (out.length < length) {
    const block = Buffer.from(x)
    block.writeUInt16BE(counter, 14) // 末尾两字节作为块计数器
    const cipher = crypto.createCipheriv('aes-128-ecb', masterKey, null)
    cipher.setAutoPadding(false)
    out = Buffer.concat([out, cipher.update(block), cipher.final()])
    counter++
  }
  return out.subarray(0, length)
}

// 从 master key/salt 派生三种会话密钥
function deriveSessionKeys(masterKey, masterSalt) {
  return {
    encKey: srtpKdf(masterKey, masterSalt, 0x00, 16),  // 会话加密密钥（AES-CM）
    authKey: srtpKdf(masterKey, masterSalt, 0x01, 20), // 会话认证密钥（HMAC-SHA1）
    saltKey: srtpKdf(masterKey, masterSalt, 0x02, 14), // 会话盐（构造每包 IV）
  }
}

// ========== RFC 3711 §4.1.1 AES-CM 每包 IV 构造 ==========
// IV = (session_salt * 2^16) XOR (SSRC * 2^64) XOR (packet_index * 2^16)
// 字节布局（16 字节大端）：
//   bytes 0..13 : session salt（14 字节）
//   bytes 4..7  : XOR SSRC（32 位）
//   bytes 8..13 : XOR packet index（48 位）
function buildIV(sessionSalt, ssrc, packetIndex) {
  const iv = Buffer.alloc(16)
  sessionSalt.copy(iv, 0, 0, 14)

  iv[4] ^= (ssrc >>> 24) & 0xff
  iv[5] ^= (ssrc >>> 16) & 0xff
  iv[6] ^= (ssrc >>> 8) & 0xff
  iv[7] ^= ssrc & 0xff

  const idx = BigInt(packetIndex)
  for (let j = 0; j < 6; j++) {
    iv[8 + j] ^= Number((idx >> BigInt(8 * (5 - j))) & 0xffn)
  }
  return iv
}

// packet_index = (ROC << 16) | SEQ
function makePacketIndex(roc, seq) {
  return (BigInt(roc) << 16n) | BigInt(seq)
}

// AES-CM 加解密（加密=解密，对称操作）
function aesCmCrypt(key, iv, data) {
  const cipher = crypto.createCipheriv('aes-128-ctr', key, iv)
  return Buffer.concat([cipher.update(data), cipher.final()])
}

// ========== protect：RTP 包 → SRTP 包 ==========
// key = deriveSessionKeys() 的输出；RTP 头保持明文，payload 加密
function protect(rtpPacket, key, salt, ssrc, seq, roc = 0) {
  const header = rtpPacket.subarray(0, RTP_HEADER_LEN)
  const payload = rtpPacket.subarray(RTP_HEADER_LEN)

  const index = makePacketIndex(roc, seq)
  const iv = buildIV(key.saltKey, ssrc, index)
  const encryptedPayload = aesCmCrypt(key.encKey, iv, payload)

  // 认证标签 = HMAC-SHA1(authKey, 头 || 密文 || ROC)，截断 10 字节
  const rocBuf = Buffer.alloc(4)
  rocBuf.writeUInt32BE(roc >>> 0, 0)
  const authTag = crypto.createHmac('sha1', key.authKey)
    .update(Buffer.concat([header, encryptedPayload, rocBuf]))
    .digest()
    .subarray(0, AUTH_TAG_LEN)

  return Buffer.concat([header, encryptedPayload, authTag])
}

// ========== unprotect：SRTP 包 → RTP 包 ==========
// 先验证 HMAC（防篡改/重放），通过后才解密
function unprotect(srtpPacket, key, salt, ssrc, seq, roc = 0) {
  const header = srtpPacket.subarray(0, RTP_HEADER_LEN)
  const encryptedPayload = srtpPacket.subarray(RTP_HEADER_LEN, srtpPacket.length - AUTH_TAG_LEN)
  const receivedTag = srtpPacket.subarray(srtpPacket.length - AUTH_TAG_LEN)

  // 1. 重算认证标签并比对（常数时间比较防时序攻击）
  const rocBuf = Buffer.alloc(4)
  rocBuf.writeUInt32BE(roc >>> 0, 0)
  const expectedTag = crypto.createHmac('sha1', key.authKey)
    .update(Buffer.concat([header, encryptedPayload, rocBuf]))
    .digest()
    .subarray(0, AUTH_TAG_LEN)

  if (!crypto.timingSafeEqual(receivedTag, expectedTag)) {
    throw new Error('SRTP 认证失败：包被篡改或密钥不匹配，整包丢弃')
  }

  // 2. 认证通过才解密
  const index = makePacketIndex(roc, seq)
  const iv = buildIV(key.saltKey, ssrc, index)
  const payload = aesCmCrypt(key.encKey, iv, encryptedPayload)

  return Buffer.concat([header, payload])
}

module.exports = {
  deriveSessionKeys,
  protect,
  unprotect,
  makePacketIndex,
  buildIV,
  RTP_HEADER_LEN,
  AUTH_TAG_LEN,
}

关键实现细节 ​

细节说明
KDF 中 label 的位置key_id = label || r（7 字节）与 14 字节 salt 右对齐异或，label 落在第 7 字节
AES-CM PRF用 aes-128-ecb 逐块加密 x + counter，拼出任意长度密钥流
IV 构造严格按 RFC 3711 §4.1.1：salt 占 bytes 0..13，SSRC 异或到 bytes 4..7，packet index 异或到 bytes 8..13
AES-CM 加解密Node 的 aes-128-ctr 与 AES-CM 完全同构（计数器从 IV 开始逐块递增）
认证顺序先验 HMAC 再解密（verify-then-decrypt），防止对密文做篡改预言机攻击
常数时间比较crypto.timingSafeEqual 防止通过比较耗时侧信道推断标签

16.3.4 main.js：端到端流程 ​

js
// main.js — 端到端验证：DTLS 握手 → SRTP 加密 → UDP 传输 → SRTP 解密
// 只用 Node.js 内置模块 crypto + dgram

const dgram = require('dgram')
const { dtlsHandshake } = require('./dtls-lite')
const { deriveSessionKeys, protect, unprotect, RTP_HEADER_LEN } = require('./srtp')

const SERVER_PORT = 5000
const SERVER_HOST = '127.0.0.1'
const SSRC = 0x11223344
const SEQ = 42

// ---------- 构造一个最小 RTP 包 ----------
function buildRtpPacket(ssrc, seq, payload) {
  const header = Buffer.alloc(RTP_HEADER_LEN)
  header[0] = 0x80                // V=2, P=0, X=0, CC=0
  header[1] = 96                  // PT=96 (动态负载类型)
  header.writeUInt16BE(seq, 2)    // 序列号
  header.writeUInt32BE(160 * 30, 4) // 时间戳（90kHz 时钟，30ms 一帧）
  header.writeUInt32BE(ssrc, 8)   // SSRC
  return Buffer.concat([header, Buffer.from(payload)])
}

async function main() {
  // ========== 第 1 步：DTLS 握手 ==========
  // 双方得到相同的密钥材料（实际通过 UDP 往返，这里同步模拟）
  const km = dtlsHandshake()

  // Client 用 client_master_* 派生「发送密钥」加密自己发的包
  const clientSendKeys = deriveSessionKeys(km.clientMasterKey, km.clientMasterSalt)
  // Server 用同一份材料派生「接收密钥」解密 client 的包（同方向同密钥）
  const serverRecvKeys = deriveSessionKeys(km.clientMasterKey, km.clientMasterSalt)

  // ========== 第 2 步：启动 UDP 服务器（SRTP 接收端） ==========
  const server = dgram.createSocket('udp4')
  const received = new Promise((resolve, reject) => {
    server.on('message', (srtpPacket) => {
      try {
        console.log('[Server] 收到 SRTP 包 (' + srtpPacket.length + ' 字节):')
        console.log('         ' + srtpPacket.toString('hex'))
        const rtp = unprotect(srtpPacket, serverRecvKeys, null, SSRC, SEQ)
        const payload = rtp.subarray(RTP_HEADER_LEN).toString()
        console.log('[Server] ✓ 认证通过 + 解密成功')
        console.log('[Server] 明文 RTP payload:', JSON.stringify(payload))
        resolve(payload)
      } catch (e) {
        reject(e)
      }
    })
  })
  await new Promise((r) => server.bind(SERVER_PORT, SERVER_HOST, r))
  console.log('[Server] UDP 监听 ' + SERVER_HOST + ':' + SERVER_PORT + '\n')

  // ========== 第 3 步：Client 构造 RTP 包 → SRTP 加密 → UDP 发送 ==========
  const rtpPacket = buildRtpPacket(SSRC, SEQ, 'hello world')
  console.log('[Client] 原始 RTP 包 (' + rtpPacket.length + ' 字节):')
  console.log('         ' + rtpPacket.toString('hex'))

  const srtpPacket = protect(rtpPacket, clientSendKeys, null, SSRC, SEQ)
  console.log('[Client] 加密后 SRTP 包 (' + srtpPacket.length + ' 字节):')
  console.log('         ' + srtpPacket.toString('hex'))
  console.log('         （前 12 字节头保持明文，payload 已加密，末尾 10 字节是认证标签）\n')

  const client = dgram.createSocket('udp4')
  client.send(srtpPacket, SERVER_PORT, SERVER_HOST)

  // ========== 第 4 步：验证结果 ==========
  const payload = await received
  if (payload === 'hello world') {
    console.log('\n✅ 端到端验证通过：密文经 UDP 传输后被正确解密为原始 payload')
  } else {
    throw new Error('解密结果与原始 payload 不一致: ' + JSON.stringify(payload))
  }

  client.close()
  server.close()
}

main().catch((e) => {
  console.error('失败:', e.message)
  process.exit(1)
})

逐字节解读输出 ​

原始 RTP 包：
80 60 00 2a 00 00 12 c0 11 22 33 44 | 68 65 6c 6c 6f 20 77 6f 72 6c 64
└─┬─┘ └──┘ └────┬────┘ └────┬────┘   └──────────┬───────────┘
V=2,PT=96 seq=42 timestamp=4800 SSRC=0x11223344  "hello world"（明文！）

加密后 SRTP 包：
80 60 00 2a 00 00 12 c0 11 22 33 44 | 3b 63 bd 81 2f 06 63 0e 3b 38 28 | 07 37 d5 8d b5 1e ad a8 bb d4
└──────────┬──────────┘              └────────────┬────────────┘         └──────────┬──────────┘
   RTP 头保持明文（12 字节，与原始完全相同）          AES-CM 密文（11 字节）              HMAC-SHA1 认证标签（10 字节）

三个可以直接观察到的结论：

  1. RTP 头逐字节未变——8060002a000012c011223344 原样出现在密文包开头，SRFU/网络设备仍能读取 SSRC 和序列号
  2. payload 面目全非——hello world 的 ASCII（68656c6c6f...）变成了随机字节
  3. 包变长 10 字节——23 → 33，正好是 80-bit 认证标签的开销（这就是 SRTP 每包的固定安全成本）

16.3.6 扩展思考 ​

1. 篡改实验：验证 HMAC 真的在起作用 ​

在 main.js 中，发送前手动翻转密文的 1 个 bit：

js
srtpPacket[15] ^= 0x01  // 篡改 payload 的第一个字节

运行结果：Server 端抛出 SRTP 认证失败：包被篡改或密钥不匹配，整包丢弃。认证标签保证任何对密文的修改都会被发现——这就是 AEAD 思想（先认证后解密）的价值。

2. 重放实验：把同一个包发两次 ​

js
client.send(srtpPacket, SERVER_PORT, SERVER_HOST)
client.send(srtpPacket, SERVER_PORT, SERVER_HOST)  // 重放

本实现没有滑动窗口，两次都会"解密成功"——这正是真实 SRTP 必须维护 64-bit 滑动窗口 + ROC 的原因（见进阶篇 §16.2.6）。有兴趣可以给 unprotect 加上窗口逻辑。

3. 与浏览器真实 DTLS-SRTP 的对照 ​

环节本实现浏览器（BoringSSL + libsrtp）
密钥交换ECDH P-256ECDHE P-256/X25519，完全一致
