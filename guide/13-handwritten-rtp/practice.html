13.3 实战案例：Node.js 手写 RTP 打包解包与统计 ​

只用 Node.js 内置的 dgram 模块，从零实现：RTP 打包/解包、RTCP SR/RR 编码解码、模拟 20ms opus 帧发送 + 15% 随机丢包、接收端统计丢包率并按 RFC 3550 公式计算 jitter、定期回送 Receiver Report。全程无第三方依赖。

13.3.1 目标与架构 ​

单进程内同时跑发送端（Sender）和接收端（Receiver），通过 127.0.0.1 上的两个 UDP 端口通信：

┌─ Sender ─────────────────────────┐     ┌─ Receiver ────────────────────────┐
│  每 20ms 生成一个 RTP 包           │     │  绑定 :5004                        │
│  payload = 模拟 opus 帧 (60~120B)  │ RTP │  解包 → 按 seq 统计丢包            │
│  15% 概率丢弃（模拟网络丢包）      │────►│  按 RFC 3550 计算 jitter          │
│  每 2s 发一次 RTCP SR ────────────┼────►│  每 1s 回送一次 RTCP RR            │
│  绑定 :5005 收 RR ◄───────────────┼─────┤  (fraction lost / 累计丢包/jitter) │
└───────────────────────────────────┘     └───────────────────────────────────┘

三个文件：

文件职责
rtp-packet.jsRTPPacket.encode() / RTPPacket.decode() —— 12 字节头的打包与解包
rtcp.jsencodeSR() / decodeSR() / encodeRR() / decodeRR() —— 含丢包统计与 jitter 字段
simulator.js收发模拟主程序，打印实时统计与最终摘要

mkdir rtp-demo && cd rtp-demo
# 依次创建下面三个文件后：
node simulator.js

13.3.2 第一步：RTP 打包 / 解包（rtp-packet.js） ​

对照 13.1 节的报文图：第 0 字节固定 0x80（V=2），第 1 字节是 M + PT，随后是大端序的 sequence（16 位）、timestamp（32 位）、SSRC（32 位）。

js
// rtp-packet.js —— RTP 打包 / 解包（纯 Node.js 内置模块）

class RTPPacket {
  /**
   * 编码 RTP 包
   * @param {Object} opts
   * @param {Buffer} opts.payload  负载数据
   * @param {number} opts.ssrc     同步源标识符
   * @param {number} opts.seq      序列号（16 位）
   * @param {number} opts.timestamp 时间戳（32 位）
   * @param {number} opts.pt       负载类型（Payload Type，7 位）
   * @param {boolean} [opts.marker=false] 标记位
   * @returns {Buffer}
   */
  static encode({ payload, ssrc, seq, timestamp, pt, marker = false }) {
    const header = Buffer.alloc(12)
    // Byte 0: V(2)=2 | P(1)=0 | X(1)=0 | CC(4)=0
    header[0] = 0x80
    // Byte 1: M(1) | PT(7)
    header[1] = (marker ? 0x80 : 0x00) | (pt & 0x7f)
    // Bytes 2-3: sequence number (16-bit BE)
    header.writeUInt16BE(seq & 0xffff, 2)
    // Bytes 4-7: timestamp (32-bit BE)
    header.writeUInt32BE(timestamp >>> 0, 4)
    // Bytes 8-11: SSRC (32-bit BE)
    header.writeUInt32BE(ssrc >>> 0, 8)
    return Buffer.concat([header, payload])
  }

  /**
   * 解码 RTP 包
   * @param {Buffer} buf
   */
  static decode(buf) {
    if (buf.length < 12) throw new Error('RTP 包长度不足 12 字节')
    const b0 = buf[0]
    const version = b0 >>> 6
    if (version !== 2) throw new Error('不是 RTP 版本 2')
    const padding = (b0 >>> 5) & 1
    const extension = (b0 >>> 4) & 1
    const csrcCount = b0 & 0x0f
    const marker = (buf[1] >>> 7) & 1
    const pt = buf[1] & 0x7f
    const seq = buf.readUInt16BE(2)
    const timestamp = buf.readUInt32BE(4)
    const ssrc = buf.readUInt32BE(8)
    // 跳过 CSRC
    const headerLen = 12 + csrcCount * 4
    // 如果有 padding，最后一个字节给出填充长度
    const payloadEnd = padding ? buf.length - buf[buf.length - 1] : buf.length
    const payload = buf.subarray(headerLen, payloadEnd)
    return { version, padding, extension, csrcCount, marker, pt, seq, timestamp, ssrc, payload }
  }
}

module.exports = { RTPPacket }

13.3.3 第二步：RTCP SR / RR（rtcp.js） ​

公共头 4 字节 + SR 发送者信息 24 字节（length = 6）；RR 含 1 个 report block，共 32 字节（length = 7）。注意 cumulative lost 是 24 位有符号数，jitter 以采样 tick 为单位。

js
// rtcp.js —— RTCP Sender Report / Receiver Report 编码与解码

const RTCP_SR = 200
const RTCP_RR = 201

/**
 * 编码 RTCP Sender Report (SR)
 */
function encodeSR({ ssrc, ntpSec, ntpFrac, rtpTimestamp, packetCount, octetCount }) {
  const buf = Buffer.alloc(28)
  buf[0] = 0x80           // V=2, P=0, RC=0
  buf[1] = RTCP_SR        // PT=200
  buf.writeUInt16BE(6, 2) // length = 28/4 - 1 = 6
  buf.writeUInt32BE(ssrc >>> 0, 4)
  buf.writeUInt32BE(ntpSec >>> 0, 8)
  buf.writeUInt32BE(ntpFrac >>> 0, 12)
  buf.writeUInt32BE(rtpTimestamp >>> 0, 16)
  buf.writeUInt32BE(packetCount >>> 0, 20)
  buf.writeUInt32BE(octetCount >>> 0, 24)
  return buf
}

function decodeSR(buf) {
  return {
    pt: buf[1],
    ssrc: buf.readUInt32BE(4),
    ntpSec: buf.readUInt32BE(8),
    ntpFrac: buf.readUInt32BE(12),
    rtpTimestamp: buf.readUInt32BE(16),
    packetCount: buf.readUInt32BE(20),
    octetCount: buf.readUInt32BE(24),
  }
}

/**
 * 编码 RTCP Receiver Report (RR)，含 1 个 report block
 */
function encodeRR({ ssrc, sourceSsrc, fractionLost, cumulativeLost, extHighestSeq, jitter, lsr = 0, dlsr = 0 }) {
  const buf = Buffer.alloc(32)
  buf[0] = 0x81           // V=2, P=0, RC=1
  buf[1] = RTCP_RR        // PT=201
  buf.writeUInt16BE(7, 2) // length = 32/4 - 1 = 7
  buf.writeUInt32BE(ssrc >>> 0, 4)
  buf.writeUInt32BE(sourceSsrc >>> 0, 8)
  buf[12] = fractionLost & 0xff
  const lost = cumulativeLost & 0xffffff
  buf[13] = (lost >>> 16) & 0xff
  buf[14] = (lost >>> 8) & 0xff
  buf[15] = lost & 0xff
  buf.writeUInt32BE(extHighestSeq >>> 0, 16)
  buf.writeUInt32BE(Math.round(jitter) >>> 0, 20)
  buf.writeUInt32BE(lsr >>> 0, 24)
  buf.writeUInt32BE(dlsr >>> 0, 28)
  return buf
}

function decodeRR(buf) {
  const cumulativeLost = (buf[13] << 16) | (buf[14] << 8) | buf[15]
  return {
    pt: buf[1],
    reporterSsrc: buf.readUInt32BE(4),
    sourceSsrc: buf.readUInt32BE(8),
    fractionLost: buf[12],
    cumulativeLost: cumulativeLost > 0x7fffff ? cumulativeLost - 0x1000000 : cumulativeLost,
    extHighestSeq: buf.readUInt32BE(16),
    jitter: buf.readUInt32BE(20),
    lsr: buf.readUInt32BE(24),
    dlsr: buf.readUInt32BE(28),
  }
}

module.exports = { RTCP_SR, RTCP_RR, encodeSR, decodeSR, encodeRR, decodeRR }

关键字段含义（对照 13.1.4 的报文图）：

  • fractionLost：自上一份 RR 以来的丢包率，用定点小数表示（实际丢包率 × 256）
  • cumulativeLost：会话开始以来的累计丢包数（应收 − 实收）
  • extHighestSeq：收到过的最大扩展序列号
  • jitter：按 RFC 3550 公式算出的抖动（tick 单位）

13.3.4 第三步：收发模拟（simulator.js） ​

要点：

  • 发送端每 20ms 发一帧，timestamp 每次 +960（48kHz × 20ms），15% 概率"丢包"（不发送）
  • 接收端用 msg[1] 区分包类型：200 = SR，否则按 RTP 解包
  • 丢包统计：应收 = maxSeq - baseSeq + 1，丢包 = 应收 - 实收
  • jitter：J = J + (|D| - J) / 16，其中 D = (R_i - R_{i-1}) - (S_i - S_{i-1})，R 用 Date.now() 换算成 tick
js
// simulator.js —— Node.js 手写 RTP 收发 + RTCP 统计模拟
// 使用纯内置模块 dgram，无外部依赖

const dgram = require('dgram')
const { RTPPacket } = require('./rtp-packet')
const rtcp = require('./rtcp')

const PORT_MEDIA = 5004    // receiver 收 RTP / SR
const PORT_RR = 5005       // sender 收 RR
const SSRC_SENDER = 0x12345678
const SSRC_RECEIVER = 0x87654321
const PT_OPUS = 111
const CLOCK_RATE = 48000          // opus 采样率
const FRAME_MS = 20               // 20 ms 一帧
const TICKS_PER_FRAME = (CLOCK_RATE * FRAME_MS) / 1000  // 960
const LOSS_RATE = 0.15            // 模拟 15% 丢包
const DURATION_MS = 6000          // 模拟 6 秒

// ===== Sender =====
const senderSock = dgram.createSocket('udp4')
let seq = 0
let timestamp = 0x01020300
let senderPackets = 0
let senderOctets = 0

senderSock.on('message', (msg) => {
  if (msg[1] !== rtcp.RTCP_RR) return
  const rr = rtcp.decodeRR(msg)
  console.log(
    `[Sender] 收到 RTCP RR: fractionLost=${(rr.fractionLost / 256 * 100).toFixed(1)}%  ` +
    `cumulativeLost=${rr.cumulativeLost}  jitter=${rr.jitter} ticks (${(rr.jitter / CLOCK_RATE * 1000).toFixed(2)} ms)`
  )
})

// 每 2 秒发一次 SR
const srTimer = setInterval(() => {
  const ntp = Date.now() / 1000 + 2208988800 // NTP 纪元偏移（1900-1970）
  const sr = rtcp.encodeSR({
    ssrc: SSRC_SENDER,
    ntpSec: Math.floor(ntp),
    ntpFrac: Math.floor((ntp % 1) * 0xffffffff),
    rtpTimestamp: timestamp,
    packetCount: senderPackets,
    octetCount: senderOctets,
  })
  senderSock.send(sr, PORT_MEDIA, '127.0.0.1')
}, 2000)

// 每 20 ms 发一帧 RTP
const senderTimer = setInterval(() => {
  const payload = Buffer.alloc(60 + Math.floor(Math.random() * 60), 0xab)
  const pkt = RTPPacket.encode({ payload, ssrc: SSRC_SENDER, seq, timestamp, pt: PT_OPUS })
  senderPackets++
  senderOctets += payload.length
  // 模拟丢包：15% 概率不发
  if (Math.random() >= LOSS_RATE) {
    senderSock.send(pkt, PORT_MEDIA, '127.0.0.1')
  }
  seq = (seq + 1) & 0xffff
  timestamp = (timestamp + TICKS_PER_FRAME) >>> 0
}, FRAME_MS)

// ===== Receiver =====
const receiverSock = dgram.createSocket('udp4')

let baseSeq = null
let maxSeq = null
let received = 0
let jitter = 0
let lastTransit = null
// 区间统计（计算 fraction lost）
let intervalReceived = 0
let intervalExpected = 0

receiverSock.on('message', (msg) => {
  if (msg[1] === rtcp.RTCP_SR) {
    const sr = rtcp.decodeSR(msg)
    console.log(`[Receiver] 收到 RTCP SR: packetCount=${sr.packetCount}  octetCount=${sr.octetCount}`)
    return
  }

  const pkt = RTPPacket.decode(msg)
  const arrivalTicks = Math.floor((Date.now() * CLOCK_RATE) / 1000)

  if (baseSeq === null) {
    baseSeq = pkt.seq
    maxSeq = pkt.seq - 1
  }
  if (pkt.seq > maxSeq) {
    intervalExpected += pkt.seq - maxSeq
    maxSeq = pkt.seq
  }
  received++
  intervalReceived++

  // jitter —— RFC 3550: J(i) = J(i-1) + (|D(i-1,i)| - J(i-1)) / 16
  const transit = arrivalTicks - pkt.timestamp
  if (lastTransit !== null) {
    const d = Math.abs(transit - lastTransit)
    jitter += (d - jitter) / 16
  }
  lastTransit = transit
})

// 每秒发一次 RTCP RR
const rrTimer = setInterval(() => {
  if (baseSeq === null) return
  const expectedTotal = maxSeq - baseSeq + 1
  const cumulativeLost = expectedTotal - received
  const fractionLost = intervalExpected > 0
    ? Math.min(255, Math.round(((intervalExpected - intervalReceived) / intervalExpected) * 256))
    : 0
  const rr = rtcp.encodeRR({
    ssrc: SSRC_RECEIVER,
    sourceSsrc: SSRC_SENDER,
    fractionLost,
    cumulativeLost,
    extHighestSeq: maxSeq,
    jitter,
  })
  receiverSock.send(rr, PORT_RR, '127.0.0.1')
  console.log(
    `[Receiver] 发送 RR: 区间丢包=${(fractionLost / 256 * 100).toFixed(1)}%  ` +
    `累计丢包=${cumulativeLost}/${expectedTotal}`
  )
  intervalReceived = 0
  intervalExpected = 0
}, 1000)

receiverSock.bind(PORT_MEDIA, () => {
  senderSock.bind(PORT_RR, () => {
    console.log(`开始模拟: ${DURATION_MS / 1000} 秒, ${FRAME_MS}ms/帧, 模拟丢包率 ${LOSS_RATE * 100}%`)
  })
})

// 6 秒后结束
setTimeout(() => {
  clearInterval(senderTimer)
  clearInterval(srTimer)
  clearInterval(rrTimer)
  const expectedTotal = maxSeq === null ? 0 : maxSeq - baseSeq + 1
  const lost = expectedTotal - received
  console.log('\n===== 统计摘要 =====')
  console.log(`发送: ${senderPackets} 包 (packetCount)`)
  console.log(`收到: ${received} 包`)
  console.log(`应收: ${expectedTotal} 包`)
  console.log(`累计丢包: ${lost} (${expectedTotal > 0 ? (lost / expectedTotal * 100).toFixed(1) : '0.0'}%)`)
  console.log(`最终 jitter: ${Math.round(jitter)} ticks ≈ ${(jitter / CLOCK_RATE * 1000).toFixed(2)} ms`)
  senderSock.close()
  receiverSock.close()
}, DURATION_MS)

从输出能验证的几件事：

  1. 丢包统计吻合：发送 286 包、收到 248 包 → 丢 38 包 ≈ 13.3%，接近模拟的 15%（样本量小，波动正常）
  2. SR 与 RR 周期符合预期：6 秒内收到 2 次 SR（每 2s 一次）、5 次 RR（每 1s 一次）
  3. jitter 量级合理：本机回环 + 定时器误差下 jitter ≈ 1ms，换算 tick / 48000 × 1000 = ms
  4. SR 与摘要自洽：SR 里的 packetCount（95、191）与接收端统计的"应收"进度一致

13.3.6 小结与练习 ​

你做到了
手写 RTP 头12 字节大端序打包/解包，V=2 校验
手写 RTCP SR/RR含 NTP 时间、packetCount、fractionLost、jitter 字段
RFC 3550 jitter 算法J = J + (|D| - J)/16，tick 单位换算毫秒
丢包统计应收 = maxSeq - baseSeq + 1，丢包 = 应收 - 实收

进一步练习（选做）：

  1. 给 RTPPacket 加上头扩展支持：X=1 时写入/解析 one-byte header（profile 0xBEDE），携带 transport-wide sequence number
  2. 实现 NACK：接收端发现 seq 跳号时，构造 PT=205/FMT=1 的包（PID + BLP）发给发送端，发送端缓存最近 200 个包并按需重发
  3. 用 extHighestSeq 支持 16 位回绕：记录回绕次数 cycles，扩展序号 = cycles | seq
