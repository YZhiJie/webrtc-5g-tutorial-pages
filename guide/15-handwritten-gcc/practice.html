15.3 实战案例：Node.js 手写简化版 GCC ​

本章节用纯 Node.js 内置模块实现一个可运行的 GCC 仿真系统。你可以直接复制代码到本地运行，观察码率从「发太多 → 被挤爆 → 被迫降速 → 慢慢恢复」的全过程。

15.3.1 项目结构 ​

创建三个文件在同一目录下：

gcc-demo/
├── gcc.js        # GCC 核心算法（Arrival / Trendline / Overuse / AIMD）
├── link-sim.js   # 虚拟链路：模拟容量变化、排队延迟、抖动
└── main.js       # 主循环：每秒跑一轮，打印码率与状态

15.3.2 gcc.js：四个核心类 ​

js
// gcc.js — 手写简化版 Google Congestion Control
// 只依赖 Node.js 内置模块

// ========== 1. ArrivalTimeFilter：到达时间滤波器 ==========
// 职责：按 tick 计算平均单向延迟(OWD)，输出相邻 tick 的 delay gradient
class ArrivalTimeFilter {
  constructor() {
    this.prevAvgOwdMs = null
  }

  // owds: 本 tick 内所有包的单向延迟样本(ms)
  // t: 当前时间戳(ms)
  computeDelayGradient(owds, t) {
    if (!owds || owds.length === 0) return null
    const avgOwd = owds.reduce((a, b) => a + b, 0) / owds.length
    let gradient = 0
    if (this.prevAvgOwdMs !== null) {
      gradient = avgOwd - this.prevAvgOwdMs
    }
    this.prevAvgOwdMs = avgOwd
    return { t, gradient }
  }
}

// ========== 2. TrendlineFilter：趋势线滤波器（最小二乘线性回归）==========
// 职责：把抖动剧烈的 delay gradient 平滑成一条趋势线，输出斜率
class TrendlineFilter {
  constructor(windowSize = 20) {
    this.windowSize = windowSize
    this.samples = [] // { t, gradient }
  }

  push(sample) {
    this.samples.push(sample)
    if (this.samples.length > this.windowSize) {
      this.samples.shift()
    }
  }

  // 最小二乘拟合 y = a + b*x，返回斜率 b（单位：ms / tick）
  getTrendlineSlope() {
    const n = this.samples.length
    if (n < 2) return 0
    let sumX = 0, sumY = 0, sumXY = 0, sumXX = 0
    for (let i = 0; i < n; i++) {
      const x = i // 用序号作为横坐标
      const y = this.samples[i].gradient
      sumX += x
      sumY += y
      sumXY += x * y
      sumXX += x * x
    }
    const denom = n * sumXX - sumX * sumX
    if (denom === 0) return 0
    return (n * sumXY - sumX * sumY) / denom
  }
}

// ========== 3. OveruseDetector：过载检测器（阈值 + 状态机）==========
// 职责：根据 trendline 斜率判断链路处于 normal / overuse / underuse 状态
class OveruseDetector {
  constructor(initialThreshold = 6.0) {
    this.threshold = initialThreshold // 初始阈值 6ms
    this.state = 'normal'             // normal | overuse | underuse
    this.overusingSince = -1
  }

  // slope: TrendlineFilter 输出的斜率(ms/tick)；nowMs: 当前时间
  detect(slope, nowMs) {
    // 持续 over 判断：斜率连续超过阈值且持续一段时间
    if (slope > this.threshold) {
      if (this.overusingSince < 0) this.overusingSince = nowMs
      if (nowMs - this.overusingSince > 100 && this.state !== 'overuse') {
        this.state = 'overuse'
      }
    } else if (slope < -this.threshold) {
      this.overusingSince = -1
      this.state = 'underuse'
    } else {
      this.overusingSince = -1
      if (this.state !== 'normal') this.state = 'normal'
    }
    this.adaptThreshold(slope)
    return this.state
  }

  adaptThreshold(slope) {
    // 简化自适应：根据斜率绝对值微调，保持在 [3, 12] ms
    const K = 0.01
    const delta = K * Math.abs(slope)
    if (Math.abs(slope) > this.threshold) {
      this.threshold = Math.min(this.threshold + delta, 12)
    } else {
      this.threshold = Math.max(this.threshold - delta, 3)
    }
  }
}

// ========== 4. AimdController：加性增 / 乘性减控制器 ==========
// 职责：根据 overuse detector 状态调整目标发送码率
class AimdController {
  constructor() {
    this.targetBitrateBps = 500_000 // 初始 500 kbps（演示时会被覆盖）
    this.minBitrateBps = 100_000    // 100 kbps
    this.maxBitrateBps = 5_000_000  // 5 Mbps
    this.additiveStep = 5_000       // 每次 +5 kbps
    this.multiplicativeFactor = 0.85
    this.lastDecreaseMs = -1
  }

  onState(state, nowMs) {
    if (state === 'overuse') {
      // 乘性减少，防止连续抖动(300ms 冷却)
      if (this.lastDecreaseMs < 0 || nowMs - this.lastDecreaseMs > 300) {
        this.targetBitrateBps = Math.max(
          this.minBitrateBps,
          Math.floor(this.targetBitrateBps * this.multiplicativeFactor)
        )
        this.lastDecreaseMs = nowMs
      }
    } else if (state === 'underuse') {
      // 加性增加
      this.targetBitrateBps = Math.min(
        this.maxBitrateBps,
        this.targetBitrateBps + this.additiveStep
      )
    } else {
      // normal：也可以缓慢增加
      this.targetBitrateBps = Math.min(
        this.maxBitrateBps,
        this.targetBitrateBps + this.additiveStep
      )
    }
    return this.targetBitrateBps
  }
}

module.exports = {
  ArrivalTimeFilter,
  TrendlineFilter,
  OveruseDetector,
  AimdController,
}

15.3.3 link-sim.js：模拟变化链路 ​

js
// link-sim.js — 模拟链路容量变化的网络环境
// 一个简化版「虚拟网卡」：给定容量(bps)，输入发包事件，输出到达时刻

const PACKET_SIZE_BYTES = 1024 // 每包 1KB

class LinkSimulator {
  constructor() {
    // 容量脚本：第几秒 → 链路容量(bps)
    this.capacitySchedule = [
      { untilSec: 5, capacityBps: 3_000_000 },  // 0-5s:  3 Mbps
      { untilSec: 10, capacityBps: 1_000_000 }, // 5-10s: 1 Mbps(骤降)
      { untilSec: Infinity, capacityBps: 5_000_000 }, // 之后: 5 Mbps(回升)
    ]
    this.queueBytes = 0 // 当前排队字节
    this.lastDrainMs = 0
  }

  capacityAt(nowMs) {
    const sec = nowMs / 1000
    for (const s of this.capacitySchedule) {
      if (sec < s.untilSec) return s.capacityBps
    }
    return this.capacitySchedule[this.capacitySchedule.length - 1].capacityBps
  }

  // 发送一个包：返回 arrivalMs（含排队延迟）
  sendPacket(sendMs) {
    // 先把从 lastDrainMs 到 sendMs 期间的队列消化掉
    const cap = this.capacityAt(sendMs)
    const drainElapsed = Math.max(0, sendMs - this.lastDrainMs)
    const drainedBytes = (cap * drainElapsed) / 1000 / 8
    this.queueBytes = Math.max(0, this.queueBytes - drainedBytes)
    this.lastDrainMs = sendMs

    // 加入队列
    this.queueBytes += PACKET_SIZE_BYTES
    // 排队延迟 = 队列长度 / 当前容量
    const queueDelayMs = (this.queueBytes * 8 * 1000) / cap
    // 基础传播延迟(模拟 5G 空口 ~20ms)
    const propagationMs = 20
    // 加入少量抖动 ±2ms
    const jitterMs = (Math.random() - 0.5) * 4
    return sendMs + propagationMs + queueDelayMs + jitterMs
  }
}

module.exports = { LinkSimulator, PACKET_SIZE_BYTES }

15.3.4 main.js：主循环与输出 ​

js
// main.js — 运行 GCC 仿真，观察码率收敛
const { ArrivalTimeFilter, TrendlineFilter, OveruseDetector, AimdController } = require('./gcc')
const { LinkSimulator, PACKET_SIZE_BYTES } = require('./link-sim')

const SIM_DURATION_MS = 15_000 // 仿真 15 秒
const TICK_MS = 1000           // 每个 tick 1 秒（与链路调度对齐）
const PROPAGATION_MS = 20      // 与 link-sim.js 保持一致

const arrivalFilter = new ArrivalTimeFilter()
const trendline = new TrendlineFilter(20)
const detector = new OveruseDetector()
const controller = new AimdController()
const link = new LinkSimulator()

// 为方便在 15 秒内观察收敛，初始码率设得略高于第一阶段容量
controller.targetBitrateBps = 3_500_000

console.log('time(s) | target(kbps) | cap(kbps) | queue(ms) | state')
console.log('-'.repeat(60))

let tickOwds = [] // 本 tick 内所有包的单向延迟
let lastReportSec = -1

for (let nowMs = 0; nowMs <= SIM_DURATION_MS; nowMs += TICK_MS) {
  // 1. 按 controller 当前码率，计算本 tick 要发多少包
  const targetBps = controller.targetBitrateBps
  const bytesPerTick = (targetBps * (TICK_MS / 1000)) / 8
  const packetsToSend = Math.max(1, Math.round(bytesPerTick / PACKET_SIZE_BYTES))

  // 2. 在本 tick 内均匀发包
  for (let i = 0; i < packetsToSend; i++) {
    const sendMs = nowMs + (i * TICK_MS) / packetsToSend
    const arrivalMs = link.sendPacket(sendMs)
    const owd = arrivalMs - sendMs // 单向延迟
    tickOwds.push(owd)
  }

  // 3. 跑 GCC 算法
  const sample = arrivalFilter.computeDelayGradient(tickOwds, nowMs)
  if (sample) trendline.push(sample)

  const slope = trendline.getTrendlineSlope()
  const state = detector.detect(slope, nowMs)
  controller.onState(state, nowMs)

  // 4. 每秒打印一次
  const sec = Math.floor(nowMs / 1000)
  if (sec !== lastReportSec && tickOwds.length > 0) {
    const avgOwd = tickOwds.reduce((a, b) => a + b, 0) / tickOwds.length
    const avgQueue = Math.max(0, avgOwd - PROPAGATION_MS)
    const cap = link.capacityAt(nowMs)
    console.log(
      `${String(sec).padStart(7)} | ` +
      `${(controller.targetBitrateBps / 1000).toFixed(0).padStart(12)} | ` +
      `${(cap / 1000).toFixed(0).padStart(9)} | ` +
      `${avgQueue.toFixed(2).padStart(9)} | ${state}`
    )
    tickOwds = []
    lastReportSec = sec
  }
}

console.log('-'.repeat(60))
console.log('仿真结束')

解读日志 ​

阶段时间发生了什么
阶段 1：轻度拥塞0-2s发送 3.5Mbps > 链路 3Mbps，排队从 85ms 涨到 421ms
阶段 2：乘性减少2-3sOveruse 触发，码率从 3505 → 2984 → 2536 kbps
阶段 3：队列排空4s码率降到 2541 < 3000，队列从 503ms 降到 423ms，进入 Underuse
阶段 4：容量骤降5s链路从 3Mbps 跌到 1Mbps，但发送端还没反应过来，排队飙到 1813ms
阶段 5：连续刹车6-9sOveruse 连续触发，码率一路乘性下降：2164 → 1839 → 1564 → 1329
阶段 6：链路恢复10-11s链路升到 5Mbps，排队迅速排空，状态变为 Underuse
阶段 7：缓慢爬升12-15s加性增加：1339 → 1344 → 1349 → 1354 → 1359 kbps（+5kbps/s）

核心观察：

  1. 乘性减少比加性增加快得多 — 2 秒从 3500 降到 2536，而 5 秒只从 1339 升到 1359
  2. 延迟是先行指标 — 排队延迟在码率调整之前就已经飙高
  3. 阈值自适应 — 系统不会一有抖动就 panic，而是等趋势确认后才行动

15.3.6 扩展思考 ​

1. 为什么 recovery 这么慢？ ​

因为我们用的是 +5 kbps/s 的保守加性步长。在生产环境中：

  • 如果链路从 1Mbps 恢复到 5Mbps，按 +5kbps/s 需要 800 秒 才能爬满
  • 实际 WebRTC 会配合 丢包反馈 做双保险：当丢包率很低时，允许更大胆的增幅

2. 怎么让仿真更真实？ ​

可以加入：

  • 乱序：让包到达顺序与发送顺序不一致
  • 随机丢包：丢包率 1-3%，观察 GCC 与丢包补偿的交互
  • 背景流量：模拟同链路有其他 TCP 流抢带宽

3. 与真实 WebRTC 的差异 ​

维度本仿真真实 WebRTC
反馈频率1 秒单次 RTCP ~100-300ms
包大小固定 1KB实际 RTP 包 200-1200 bytes
阈值自适应简化版更复杂的指数加权
编码器联动无编码器真正按码率编码
