11.3 实战案例：Node.js 手写 STUN Server 与 Client ​

只用 Node.js 内置模块（dgram、crypto），从零实现一个可运行的 STUN 协议栈：编码/解码、Binding 请求/响应、XOR-MAPPED-ADDRESS、MESSAGE-INTEGRITY、FINGERPRINT。运行后你能亲眼看到客户端拿到自己的公网映射地址。

11.3.1 环境准备 ​

无需安装任何依赖，Node.js ≥ 18 即可：

bash
node --version   # v18.0.0+，建议 v20+

创建项目目录：

bash
mkdir stun-demo && cd stun-demo

11.3.2 协议层：stun-protocol.js ​

负责 STUN 报文的编码（Encode）、解码（Decode）、XOR 地址换算、CRC32 指纹。

js
const crypto = require('crypto');
const zlib = require('zlib');

const MAGIC_COOKIE = 0x2112A442;
const MAGIC_COOKIE_BUF = Buffer.from([0x21, 0x12, 0xa4, 0x42]);
const BINDING_REQUEST = 0x0001;
const BINDING_SUCCESS = 0x0101;

function encodeAttrs(attrs) {
  let len = 0;
  for (const a of attrs) {
    const pad = (4 - (a.value.length % 4)) % 4;
    len += 4 + a.value.length + pad;
  }
  const buf = Buffer.allocUnsafe(len);
  let off = 0;
  for (const a of attrs) {
    buf.writeUInt16BE(a.type, off);
    buf.writeUInt16BE(a.value.length, off + 2);
    a.value.copy(buf, off + 4);
    off += 4 + a.value.length;
    const pad = (4 - (a.value.length % 4)) % 4;
    if (pad) buf.fill(0, off, off + pad);
    off += pad;
  }
  return buf;
}

function makeHeader(msgType, bodyLen, tid) {
  const buf = Buffer.allocUnsafe(20);
  buf.writeUInt16BE(msgType, 0);
  buf.writeUInt16BE(bodyLen, 2);
  MAGIC_COOKIE_BUF.copy(buf, 4);
  if (typeof tid === 'string') Buffer.from(tid, 'hex').copy(buf, 8);
  else tid.copy(buf, 8);
  return buf;
}

function buildRequest(tid) {
  return makeHeader(BINDING_REQUEST, 0, tid);
}

function buildResponse(tid, ip, port, key) {
  const p = ip.split('.').map(Number);
  const addrNum = (p[0] << 24) | (p[1] << 16) | (p[2] << 8) | p[3];

  const xorAddr = Buffer.allocUnsafe(8);
  xorAddr[0] = 0; xorAddr[1] = 0x01;
  xorAddr.writeUInt16BE(port ^ (MAGIC_COOKIE >>> 16), 2);
  xorAddr.writeUInt32BE(addrNum ^ MAGIC_COOKIE, 4);

  const mapAddr = Buffer.allocUnsafe(8);
  mapAddr[0] = 0; mapAddr[1] = 0x01;
  mapAddr.writeUInt16BE(port, 2);
  mapAddr.writeUInt32BE(addrNum, 4);

  let body = encodeAttrs([
    { type: 0x0020, value: xorAddr },
    { type: 0x0001, value: mapAddr }
  ]);

  if (key) {
    const bodyWithMi = body.length + 24;
    const miHeader = Buffer.from([0x00, 0x08, 0x00, 0x14]);
    const placeholder = Buffer.concat([makeHeader(BINDING_SUCCESS, bodyWithMi, tid), body, miHeader, Buffer.alloc(20)]);
    const hmac = crypto.createHmac('sha1', key).update(placeholder).digest();
    body = Buffer.concat([body, encodeAttrs([{ type: 0x0008, value: hmac }])]);
  }

  const bodyWithFp = body.length + 8;
  const fpHeader = Buffer.from([0x80, 0x28, 0x00, 0x04]);
  const placeholder = Buffer.concat([makeHeader(BINDING_SUCCESS, bodyWithFp, tid), body, fpHeader, Buffer.alloc(4)]);
  const fpVal = Buffer.allocUnsafe(4);
  fpVal.writeUInt32BE((zlib.crc32(placeholder) ^ 0x5354554e) >>> 0, 0);
  body = Buffer.concat([body, encodeAttrs([{ type: 0x8028, value: fpVal }])]);

  return Buffer.concat([makeHeader(BINDING_SUCCESS, body.length, tid), body]);
}

function decodeStun(buf) {
  if (buf.length < 20) return null;
  const msgType = buf.readUInt16BE(0);
  const bodyLen = buf.readUInt16BE(2);
  const magicCookie = buf.readUInt32BE(4);
  const transactionId = buf.subarray(8, 20);
  if (magicCookie !== MAGIC_COOKIE) return null;
  if (buf.length !== 20 + bodyLen) return null;

  const attrs = [];
  let off = 20;
  const end = 20 + bodyLen;
  while (off < end) {
    if (off + 4 > end) break;
    const type = buf.readUInt16BE(off);
    const len = buf.readUInt16BE(off + 2);
    const padded = len + ((4 - (len % 4)) % 4);
    if (off + 4 + padded > end) break;
    attrs.push({ type, length: len, value: buf.subarray(off + 4, off + 4 + len) });
    off += 4 + padded;
  }
  return { msgType, bodyLen, transactionId, attrs };
}

function decodeXorMapped(value) {
  const xPort = value.readUInt16BE(2);
  const port = xPort ^ (MAGIC_COOKIE >>> 16);
  const xAddr = value.readUInt32BE(4);
  const addrNum = xAddr ^ MAGIC_COOKIE;
  const ip = [(addrNum >>> 24) & 0xff, (addrNum >>> 16) & 0xff, (addrNum >>> 8) & 0xff, addrNum & 0xff].join('.');
  return { family: value[1], port, ip };
}

function hexDump(buf) {
  for (let i = 0; i < buf.length; i += 16) {
    const line = buf.subarray(i, i + 16);
    const hex = Array.from(line).map(b => b.toString(16).padStart(2, '0')).join(' ');
    const ascii = Array.from(line).map(b => (b >= 32 && b < 127) ? String.fromCharCode(b) : '.').join('');
    console.log(`${i.toString(16).padStart(4, '0')}  ${hex.padEnd(48, ' ')}  ${ascii}`);
  }
}

module.exports = {
  MAGIC_COOKIE, BINDING_REQUEST, BINDING_SUCCESS,
  buildRequest, buildResponse, decodeStun, decodeXorMapped, hexDump
};

核心要点：

  • encodeAttrs 处理 TLV 编码和 4 字节对齐
  • buildResponse 中 MESSAGE-INTEGRITY 必须在 FINGERPRINT 之前计算，且计算时要用占位符把头部长度预先算上
  • decodeStun 先校验 Magic Cookie（0x2112A442），再按 TLV 循环解析属性

11.3.3 服务端：stun-server.js ​

用 dgram 创建 UDP socket，收到 Binding Request 后构造 Success Response 返回。

js
const dgram = require('dgram');
const { buildResponse, decodeStun } = require('./stun-protocol');

const PORT = 3478;
const KEY = Buffer.from('demo-key-12345');

const server = dgram.createSocket('udp4');

server.on('message', (msg, rinfo) => {
  const parsed = decodeStun(msg);
  if (!parsed) {
    console.log('Ignored non-STUN packet from', rinfo.address + ':' + rinfo.port);
    return;
  }
  console.log('Binding Request from', rinfo.address + ':' + rinfo.port,
              'tid=' + parsed.transactionId.toString('hex'));

  const response = buildResponse(parsed.transactionId, rinfo.address, rinfo.port, KEY);
  server.send(response, rinfo.port, rinfo.address, (err) => {
    if (err) console.error('Send error:', err);
    else console.log('Sent Binding Success Response (' + response.length + ' bytes)');
  });
});

server.bind(PORT, () => {
  console.log('STUN server listening on UDP port', PORT);
});

注意：rinfo.address / rinfo.port 就是客户端的公网映射地址（在真实 NAT 场景下）。服务器把它填入 XOR-MAPPED-ADDRESS 和 MAPPED-ADDRESS 返回。

11.3.4 客户端：stun-client.js ​

发送 Binding Request，收到响应后打印 hex dump 并解析出公网映射地址。

js
const dgram = require('dgram');
const crypto = require('crypto');
const { buildRequest, decodeStun, decodeXorMapped, hexDump } = require('./stun-protocol');

const SERVER_HOST = process.argv[2] || '127.0.0.1';
const SERVER_PORT = parseInt(process.argv[3] || '3478', 10);

const client = dgram.createSocket('udp4');
const tid = crypto.randomBytes(12);
const request = buildRequest(tid);

console.log('Sending Binding Request to', SERVER_HOST + ':' + SERVER_PORT);
console.log('Transaction ID:', tid.toString('hex'));
console.log('\n--- Request Hex Dump ---');
hexDump(request);

client.send(request, SERVER_PORT, SERVER_HOST);

const timer = setTimeout(() => {
  console.log('\nTimeout: no response received');
  client.close();
}, 5000);

client.on('message', (msg) => {
  clearTimeout(timer);
  console.log('\n--- Response Hex Dump ---');
  hexDump(msg);

  const parsed = decodeStun(msg);
  if (!parsed) {
    console.log('\nInvalid STUN response');
    client.close();
    return;
  }

  console.log('\nParsed STUN Response:');
  console.log('  Message Type:   0x' + parsed.msgType.toString(16).padStart(4, '0'));
  console.log('  Body Length:    ' + parsed.bodyLen);
  console.log('  Transaction ID: ' + parsed.transactionId.toString('hex'));

  for (const attr of parsed.attrs) {
    const typeStr = '0x' + attr.type.toString(16).padStart(4, '0');
    if (attr.type === 0x0020) {
      const addr = decodeXorMapped(attr.value);
      console.log('\n  [XOR-MAPPED-ADDRESS ' + typeStr + ']');
      console.log('    Public Mapping: ' + addr.ip + ':' + addr.port);
      console.log('    ✅ This is what the STUN server sees as your address');
    } else if (attr.type === 0x0001) {
      const port = attr.value.readUInt16BE(2);
      const ip = [attr.value[4], attr.value[5], attr.value[6], attr.value[7]].join('.');
      console.log('\n  [MAPPED-ADDRESS ' + typeStr + ']');
      console.log('    Plain Address:  ' + ip + ':' + port);
    } else if (attr.type === 0x0008) {
      console.log('\n  [MESSAGE-INTEGRITY ' + typeStr + ']');
      console.log('    HMAC-SHA1:      ' + attr.value.toString('hex').slice(0, 16) + '...');
    } else if (attr.type === 0x8028) {
      console.log('\n  [FINGERPRINT ' + typeStr + ']');
      console.log('    CRC32 Value:    0x' + attr.value.readUInt32BE(0).toString(16));
    }
  }

  client.close();
});

11.3.7 观察要点 ​

Hex Dump 对照解读 ​

01 01 00 38 21 12 a4 42 ab e8 ec e2 c2 59 e3 98 cd 54 75 d6
│  │  │  │  │           │
│  │  │  │  │           └─ Transaction ID (12 bytes)
│  │  │  │  └─ Magic Cookie 0x2112A442
│  │  │  └─ Body Length = 0x0038 = 56 bytes
│  │  └─ Message Type = 0x0101 (Binding Success Response)
│  └─ 前 2 位固定为 0
└─ 高 2 位 Class = 01，Method = 0x001，组合后 0x0101

00 20 00 08 00 01 f4 6a 5e 12 a4 43
│  │  │  │  │  │
│  │  │  │  │  └─ X-Address = 0x5e12a443
│  │  │  │  │      实际 IP = 0x5e12a443 XOR 0x2112a442 = 0x7f000001 = 127.0.0.1
│  │  │  │  └─ X-Port = 0xf46a
│  │  │  │      实际端口 = 0xf46a XOR 0x2112 = 0xd578 = 54648
│  │  │  └─ Family = 0x01 (IPv4)
│  │  └─ Length = 0x0008 = 8
│  └─ Type = 0x0020 (XOR-MAPPED-ADDRESS)

验证 XOR 解码 ​

js
// 端口验证
0xf46a ^ 0x2112 = 0xd578 = 54648 ✓

// IP 验证
0x5e12a443 ^ 0x2112a442 = 0x7f000001 = 127.0.0.1 ✓

消息长度验证 ​

属性头 + Value长度
XOR-MAPPED-ADDRESS4 + 812
MAPPED-ADDRESS4 + 812
MESSAGE-INTEGRITY4 + 2024
FINGERPRINT4 + 48
合计 Body56

加上 20 字节头部 = 76 字节，与输出完全一致。

11.3.8 进阶实验 ​

  1. 抓包验证：用 Wireshark/tcpdump 抓 UDP 3478 端口，过滤 stun，对比你的实现与标准 STUN 报文格式。

  2. 真实 NAT 测试：把 stun-server.js 部署到公网服务器，客户端运行后会看到自己的真实公网映射地址，而不是 127.0.0.1。

  3. 篡改检测：修改 buildResponse 中的某个字节再发送，客户端用 FINGERPRINT 或 MESSAGE-INTEGRITY 校验时会报错（可尝试手动添加校验逻辑）。

  4. RFC 3489 兼容：把 MAGIC_COOKIE_BUF 换成 16 字节随机 Transaction ID，观察解码器如何拒绝没有 Magic Cookie 的旧报文。
