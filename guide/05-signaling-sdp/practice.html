5.3 实战案例：完整的 Node.js + WebSocket 信令服务 ​

从零搭建一个可直接用于 WebRTC 音视频通话/直播的信令服务器，并配套前端客户端。

5.3.3 server.js（完整代码） ​

js
// server.js —— WebRTC 信令服务器
import { WebSocketServer } from 'ws'

const PORT = 8080
const wss = new WebSocketServer({ port: PORT })

// 房间管理
/** @type {Map<string, Map<string, import('ws').WebSocket>>} */
const rooms = new Map()
/** @type {Map<import('ws').WebSocket, { roomId: string, userId: string }>} */
const clients = new Map()

wss.on('connection', (ws) => {
  ws.isAlive = true
  
  ws.on('pong', () => { ws.isAlive = true })
  
  ws.on('message', (raw) => {
    let msg
    try { msg = JSON.parse(raw.toString()) } catch { return }
    
    const info = clients.get(ws)
    
    switch (msg.type) {
      case 'join': {
        const { roomId, userId } = msg
        if (!roomId || !userId) return
        if (!rooms.has(roomId)) rooms.set(roomId, new Map())
        rooms.get(roomId).set(userId, ws)
        clients.set(ws, { roomId, userId })
        
        // 告诉新人现有成员列表
        const members = Array.from(rooms.get(roomId).keys()).filter(u => u !== userId)
        ws.send(JSON.stringify({ type: 'room-members', members }))
        
        // 通知现有成员有人加入
        const notify = JSON.stringify({ type: 'peer-joined', userId })
        for (const [u, peer] of rooms.get(roomId)) {
          if (u !== userId && peer.readyState === 1) peer.send(notify)
        }
        break
      }
      
      case 'offer':
      case 'answer':
      case 'ice-candidate':
      case 'bye': {
        if (!info?.roomId) return
        // 精确路由
        const room = rooms.get(info.roomId)
        const target = room?.get(msg.to)
        if (target && target.readyState === 1) {
          target.send(JSON.stringify({ ...msg, from: info.userId }))
        }
        // bye 时同步清理
        if (msg.type === 'bye') {
          room?.delete(info.userId)
        }
        break
      }
    }
  })
  
  ws.on('close', () => {
    const info = clients.get(ws)
    if (info?.roomId) {
      const room = rooms.get(info.roomId)
      room?.delete(info.userId)
      if (room?.size === 0) rooms.delete(info.roomId)
      // 通知其他人
      const notify = JSON.stringify({ type: 'peer-left', userId: info.userId })
      for (const [, peer] of (room || [])) {
        if (peer.readyState === 1) peer.send(notify)
      }
    }
    clients.delete(ws)
  })
})

// 心跳
setInterval(() => {
  wss.clients.forEach(ws => {
    if (!ws.isAlive) return ws.terminate()
    ws.isAlive = false
    ws.ping()
  })
}, 30_000)

console.log(`✅ WebRTC Signaling Server on ws://localhost:${PORT}`)
console.log(`   支持房间管理、Offer/Answer 路由、ICE Trickle、心跳保活`)

5.3.4 前端 client.js 片段 ​

js
// ========== 初始化 ==========
const CONFIG = {
  signalingUrl: 'ws://localhost:8080',
  iceServers: [
    { urls: 'stun:stun.l.google.com:19302' },
    // 生产别忘了 TURN
  ]
}

const userId = 'user-' + Math.random().toString(36).slice(2, 8)
const roomId = prompt('输入房间号', 'demo-room') || 'demo-room'
const localVideo = document.getElementById('local-video')
const remoteVideo = document.getElementById('remote-video')

let ws, pc, localStream

// ========== 信令连接 ==========
function connect() {
  ws = new WebSocket(`${CONFIG.signalingUrl}?room=${roomId}&user=${userId}`)
  
  ws.onopen = () => {
    console.log('✅ 信令已连接')
    ws.send(JSON.stringify({ type: 'join', roomId, userId }))
  }
  
  ws.onclose = () => setTimeout(connect, 2000)  // 重连
  
  ws.onmessage = async (evt) => {
    const msg = JSON.parse(evt.data)
    console.log('📨', msg.type)
    
    switch (msg.type) {
      case 'room-members':
        // 收到房间已有成员，主动对每个成员发 Offer
        msg.members.forEach(targetUserId => createPeerConnection(targetUserId, true))
        break
      
      case 'peer-joined':
        // 新人加入，为他创建连接（但不发 Offer，等他发过来）
        createPeerConnection(msg.userId, false)
        break
      
      case 'offer':
        // 收到 Offer → 生成 Answer
        await pc.setRemoteDescription(new RTCSessionDescription(msg.sdp))
        const answer = await pc.createAnswer()
        await pc.setLocalDescription(answer)
        send({ type: 'answer', sdp: answer.sdp, to: msg.from })
        break
      
      case 'answer':
        await pc.setRemoteDescription(new RTCSessionDescription(msg.sdp))
        break
      
      case 'ice-candidate':
        await pc.addIceCandidate(new RTCIceCandidate(msg.candidate))
        break
      
      case 'peer-left':
        remoteVideo.srcObject = null
        break
    }
  }
}

function send(msg) { ws.send(JSON.stringify(msg)) }

// ========== WebRTC 连接 ==========
async function createPeerConnection(targetUserId, isInitiator) {
  if (pc) pc.close()
  
  pc = new RTCPeerConnection({ iceServers: CONFIG.iceServers })
  
  // 添加本地流
  localStream.getTracks().forEach(track => pc.addTrack(track, localStream))
  
  pc.addEventListener('track', (e) => {
    remoteVideo.srcObject = e.streams[0]
  })
  
  pc.addEventListener('icecandidate', (e) => {
    if (e.candidate) send({ type: 'ice-candidate', candidate: e.candidate, to: targetUserId })
  })
  
  pc.addEventListener('connectionstatechange', () => {
    console.log('🔗 PeerConnection 状态:', pc.connectionState)
  })
  
  if (isInitiator) {
    const offer = await pc.createOffer()
    await pc.setLocalDescription(offer)
    send({ type: 'offer', sdp: offer.sdp, to: targetUserId })
  }
}

// ========== 启动 ==========
async function init() {
  localStream = await navigator.mediaDevices.getUserMedia({ video: true, audio: true })
  localVideo.srcObject = localStream
  connect()
}

init().catch(err => alert('初始化失败: ' + err.message))

5.3.6 常见问题排查 ​

现象排查
Offer 发出但远端没收到信令服务器日志看有没有转发；检查 msg.to 是否是正确的 userId
ICE 状态一直 checkingSTUN/TURN 配置问题；用 pc.getConfiguration() 检查
setRemoteDescription 报错SDP 格式损坏；检查是否转义了换行
两端都能 setRemoteDescription 但没画面track 没 add 或没监听 ontrack；检查 pc.getSenders()
