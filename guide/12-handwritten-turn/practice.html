12.3 实战案例：Node.js 手写 mini TURN 服务器 ​

只用 Node.js 内置模块 dgram 和 crypto，零依赖手写一个 mini TURN 服务器 + 客户端，完整跑通 RFC 5766 的核心事务：Allocate（401 质询 + 长期凭证）→ CreatePermission → Data/Send indication → ChannelBind → ChannelData → Refresh 释放。全部代码已在本机实测通过。

12.3.1 目标与架构 ​

  turn-client.js              mini-turn-server.js              peer（模拟对端）
  ┌─────────────┐             ┌──────────────────────┐         ┌─────────────┐
  │ UDP socket  │── TURN 消息─►│ 控制口 :3478          │         │ UDP socket  │
  │ :随机端口    │             │                      │         │ :随机端口    │
  │             │◄─Data ind./─│ relay 口 :随机端口    │◄─裸 UDP─│             │
  │             │  ChannelData│ （= relayed address） │────裸 UDP──►│          │
  └─────────────┘             └──────────────────────┘         └─────────────┘

我们将验证的完整流程：

步骤动作期望结果
1Allocate（无凭证）401 + REALM/NONCE
2Allocate（带 MESSAGE-INTEGRITY）success，拿到 relayed address，租约 600s
3CreatePermission（peer IP）success
4peer → relayed addressclient 收到 Data indication
5ChannelBind（0x4001 ↔ peer）success
6peer 再发client 收到 4 字节头的 ChannelData
7client 发 Send indicationpeer 收到数据（源端口 = relay 口）
8client 发 ChannelDatapeer 收到数据
9Refresh（LIFETIME=0）分配释放

12.3.2 mini-turn-server.js ​

保存为 mini-turn-server.js：

js
/**
 * mini-turn-server.js
 * 手写 mini TURN 服务器（RFC 5766 子集实现）
 * 只用 Node.js 内置模块：dgram（UDP）/ crypto（MD5、HMAC-SHA1）
 */
'use strict'
const dgram = require('dgram')
const crypto = require('crypto')

// ==================== 协议常量 ====================
const HOST = '127.0.0.1'
const PORT = 3478
const REALM = 'mini-turn'
const USERS = { webrtc: 'webrtc2024' }   // 用户名 -> 明文密码（demo）
const ALLOCATION_LIFETIME = 600          // 分配租约 600 秒
const PERMISSION_LIFETIME = 300          // 权限 300 秒（5 分钟）
const MAGIC_COOKIE = 0x2112a442

// TURN 方法号（RFC 5766 §13）
const M = {
  Allocate: 0x003,
  Refresh: 0x004,
  Send: 0x006,           // Send indication
  Data: 0x007,           // Data indication
  CreatePermission: 0x008,
  ChannelBind: 0x009,
}
// 消息类别
const CLS = { Request: 0, Indication: 1, Success: 2, Error: 3 }
// 属性号
const A = {
  Username: 0x0006,
  MessageIntegrity: 0x0008,
  ErrorCode: 0x0009,
  ChannelNumber: 0x000c,
  Lifetime: 0x000d,
  XorPeerAddress: 0x0012,
  Data: 0x0013,
  Realm: 0x0014,
  Nonce: 0x0015,
  XorRelayedAddress: 0x0016,
  RequestedTransport: 0x0019,
  XorMappedAddress: 0x0020,
}

// ==================== STUN/TURN 编解码 ====================
// 消息类型 = 方法号与类别按位穿插编码（RFC 5389 §6）
function encodeType(method, klass) {
  return (method & 0x000f) |
    ((method & 0x0070) << 1) |
    ((method & 0x0f80) << 2) |
    ((klass & 0x01) << 4) |
    ((klass & 0x02) << 7)
}
function decodeType(t) {
  return {
    method: (t & 0x000f) | ((t & 0x00e0) >> 1) | ((t & 0x3e00) >> 2),
    klass: ((t & 0x0010) >> 4) | ((t & 0x0100) >> 7),
  }
}
const pad4 = (n) => (n + 3) & ~3

// 组装一条 STUN/TURN 消息（attrs: [{type, value:Buffer}]）
function buildMessage(type, txnIdHex, attrs) {
  let body = Buffer.alloc(0)
  for (const a of attrs) {
    const head = Buffer.alloc(4)
    head.writeUInt16BE(a.type, 0)
    head.writeUInt16BE(a.value.length, 2)
    body = Buffer.concat([body, head, a.value, Buffer.alloc(pad4(a.value.length) - a.value.length)])
  }
  const head = Buffer.alloc(20)
  head.writeUInt16BE(type, 0)
  head.writeUInt16BE(body.length, 2)
  head.writeUInt32BE(MAGIC_COOKIE, 4)
  Buffer.from(txnIdHex, 'hex').copy(head, 8)
  return Buffer.concat([head, body])
}

function parseMessage(msg) {
  const m = decodeType(msg.readUInt16BE(0))
  const len = msg.readUInt16BE(2)
  m.txnId = msg.subarray(8, 20).toString('hex')
  m.attrs = []
  let off = 20
  while (off + 4 <= 20 + len) {
    const t = msg.readUInt16BE(off)
    const l = msg.readUInt16BE(off + 2)
    m.attrs.push({ type: t, value: msg.subarray(off + 4, off + 4 + l) })
    off += 4 + pad4(l)
  }
  m.raw = msg
  return m
}
const getAttr = (attrs, type) => {
  const a = attrs.find((x) => x.type === type)
  return a ? a.value : null
}

// XOR 地址：端口与 IP 分别与魔数异或（防 NAT 改写载荷里的明文地址）
function encodeXorAddress(addr, port) {
  const buf = Buffer.alloc(8)
  buf[1] = 0x01 // family = IPv4
  buf.writeUInt16BE(port ^ 0x2112, 2)
  const ip = addr.split('.').map(Number)
  const ck = [0x21, 0x12, 0xa4, 0x42]
  for (let i = 0; i < 4; i++) buf[4 + i] = ip[i] ^ ck[i]
  return buf
}
function decodeXorAddress(buf) {
  const ck = [0x21, 0x12, 0xa4, 0x42]
  return {
    port: buf.readUInt16BE(2) ^ 0x2112,
    addr: [0, 1, 2, 3].map((i) => buf[4 + i] ^ ck[i]).join('.'),
  }
}

const u32 = (n) => { const b = Buffer.alloc(4); b.writeUInt32BE(n, 0); return b }

function encodeErrorCode(code, reason) {
  return Buffer.concat([
    Buffer.from([0, 0, Math.floor(code / 100), code % 100]),
    Buffer.from(reason),
  ])
}

// ==================== 长期凭证（LT Cred）====================
// key = MD5(username:realm:password)
const longTermKey = (u, r, p) =>
  crypto.createHash('md5').update(`${u}:${r}:${p}`).digest()

// 给消息追加 MESSAGE-INTEGRITY（HMAC-SHA1）
function signMessage(msg, key) {
  // 长度字段先调大 24（MI 属性自身占 24 字节），再对整段做 HMAC
  msg.writeUInt16BE(msg.readUInt16BE(2) + 24, 2)
  const mac = crypto.createHmac('sha1', key).update(msg).digest()
  const mi = Buffer.alloc(24)
  mi.writeUInt16BE(A.MessageIntegrity, 0)
  mi.writeUInt16BE(20, 2)
  mac.copy(mi, 4)
  return Buffer.concat([msg, mi])
}
function verifyMessageIntegrity(msg, key) {
  const total = 20 + msg.readUInt16BE(2)
  let off = 20
  while (off + 4 <= total) {
    const t = msg.readUInt16BE(off)
    const l = msg.readUInt16BE(off + 2)
    if (t === A.MessageIntegrity) {
      const prefix = Buffer.from(msg.subarray(0, off)) // 拷贝，避免改原包
      prefix.writeUInt16BE((off - 20) + 24, 2)
      const mac = crypto.createHmac('sha1', key).update(prefix).digest()
      return mac.equals(msg.subarray(off + 4, off + 4 + l))
    }
    off += 4 + pad4(l)
  }
  return false
}

// ==================== 服务器主体 ====================
const server = dgram.createSocket('udp4')
const sessions = new Map() // clientKey -> session（一个 Allocate 一个 session）
const nonces = new Map()   // clientKey -> 当前有效 nonce

const clientKeyOf = (rinfo) => `${rinfo.address}:${rinfo.port}`
const reply = (rinfo, buf) => server.send(buf, rinfo.port, rinfo.address)

function sendError(rinfo, txnId, method, code, reason, extraAttrs, key) {
  let msg = buildMessage(encodeType(method, CLS.Error), txnId, [
    { type: A.ErrorCode, value: encodeErrorCode(code, reason) },
    ...extraAttrs,
  ])
  if (key) msg = signMessage(msg, key)
  reply(rinfo, msg)
  console.log(`[server] -> ${code} ${reason}（发给 ${clientKeyOf(rinfo)}）`)
}

server.on('message', (msg, rinfo) => {
  // 高两位 01（0x4000-0x7FFF）的是 ChannelData，不是 STUN 消息
  const first16 = msg.readUInt16BE(0)
  if (first16 >= 0x4000 && first16 < 0x8000) return onChannelData(msg, rinfo)

  let m
  try { m = parseMessage(msg) } catch { return }
  if (m.method === M.Allocate && m.klass === CLS.Request) return onAllocate(m, rinfo)
  if (m.method === M.CreatePermission && m.klass === CLS.Request) return onCreatePermission(m, rinfo)
  if (m.method === M.ChannelBind && m.klass === CLS.Request) return onChannelBind(m, rinfo)
  if (m.method === M.Send && m.klass === CLS.Indication) return onSendIndication(m, rinfo)
  if (m.method === M.Refresh && m.klass === CLS.Request) return onRefresh(m, rinfo)
})

// ---------- Allocate ----------
function onAllocate(m, rinfo) {
  const ckey = clientKeyOf(rinfo)
  console.log(`[server] <- Allocate request（来自 ${ckey}）`)
  const username = getAttr(m.attrs, A.Username)
  const realm = getAttr(m.attrs, A.Realm)
  const nonce = getAttr(m.attrs, A.Nonce)
  const mi = getAttr(m.attrs, A.MessageIntegrity)

  // 第一次：无凭证 -> 401 + REALM + NONCE 质询
  if (!username || !realm || !nonce || !mi) {
    const n = crypto.randomBytes(12).toString('hex')
    nonces.set(ckey, n)
    return sendError(rinfo, m.txnId, M.Allocate, 401, 'Unauthorized', [
      { type: A.Realm, value: Buffer.from(REALM) },
      { type: A.Nonce, value: Buffer.from(n) },
    ])
  }

  // 第二次：校验长期凭证
  const user = username.toString()
  const password = USERS[user]
  const key = password ? longTermKey(user, REALM, password) : null
  if (!key || nonce.toString() !== nonces.get(ckey) || !verifyMessageIntegrity(m.raw, key)) {
    return sendError(rinfo, m.txnId, M.Allocate, 401, 'Unauthorized', [
      { type: A.Realm, value: Buffer.from(REALM) },
      { type: A.Nonce, value: Buffer.from(nonces.get(ckey)) },
    ])
  }

  // 凭证通过：为该分配开一个独立的 UDP 中继口（relayed address）
  const relay = dgram.createSocket('udp4')
  relay.bind(0, HOST, () => {
    const { address: raddr, port: rport } = relay.address()
    const session = {
      key, relay, client: rinfo,
      relayAddr: raddr, relayPort: rport,
      permissions: new Map(),   // peerIp -> 过期时间戳
      channels: new Map(),      // channelNum -> {addr, port}
      peerChannels: new Map(),  // 'ip:port' -> channelNum
      expireAt: Date.now() + ALLOCATION_LIFETIME * 1000,
    }
    sessions.set(ckey, session)
    relay.on('message', (data, peer) => onRelayData(session, data, peer))

    const ok = buildMessage(encodeType(M.Allocate, CLS.Success), m.txnId, [
      { type: A.XorRelayedAddress, value: encodeXorAddress(raddr, rport) },
      { type: A.Lifetime, value: u32(ALLOCATION_LIFETIME) },
      { type: A.XorMappedAddress, value: encodeXorAddress(rinfo.address, rinfo.port) },
    ])
    reply(rinfo, signMessage(ok, key))
    console.log(`[server] -> Allocate success，relayed address = ${raddr}:${rport}（租约 ${ALLOCATION_LIFETIME}s）`)
  })
}

// ---------- CreatePermission ----------
function onCreatePermission(m, rinfo) {
  const ckey = clientKeyOf(rinfo)
  const session = sessions.get(ckey)
  console.log(`[server] <- CreatePermission（来自 ${ckey}）`)
  if (!session) return sendError(rinfo, m.txnId, M.CreatePermission, 437, 'Allocation Mismatch', [], null)
  if (!verifyMessageIntegrity(m.raw, session.key)) {
    return sendError(rinfo, m.txnId, M.CreatePermission, 401, 'Unauthorized', [
      { type: A.Realm, value: Buffer.from(REALM) },
      { type: A.Nonce, value: Buffer.from(nonces.get(ckey)) },
    ])
  }
  // 权限只绑定 peer 的 IP（不含端口），可带多个 XOR-PEER-ADDRESS
  for (const a of m.attrs.filter((x) => x.type === A.XorPeerAddress)) {
    const peer = decodeXorAddress(a.value)
    session.permissions.set(peer.addr, Date.now() + PERMISSION_LIFETIME * 1000)
    console.log(`[server]    允许 peer IP ${peer.addr}（${PERMISSION_LIFETIME}s 有效）`)
  }
  const ok = buildMessage(encodeType(M.CreatePermission, CLS.Success), m.txnId, [])
  reply(rinfo, signMessage(ok, session.key))
  console.log('[server] -> CreatePermission success')
}

// ---------- Send indication（client -> peer）----------
function onSendIndication(m, rinfo) {
  const session = sessions.get(clientKeyOf(rinfo))
  if (!session) return
  const peer = decodeXorAddress(getAttr(m.attrs, A.XorPeerAddress))
  const data = getAttr(m.attrs, A.Data)
  const perm = session.permissions.get(peer.addr)
  if (!perm || perm < Date.now()) {
    return console.log(`[server]    ${peer.addr} 无权限，丢弃 Send indication`)
  }
  session.relay.send(data, peer.port, peer.addr)
  console.log(`[server]    Send indication：经 relay 口转发 ${data.length}B -> ${peer.addr}:${peer.port}`)
}

// ---------- relay 口收到 peer 数据（peer -> client）----------
function onRelayData(session, data, peer) {
  const perm = session.permissions.get(peer.address)
  if (!perm || perm < Date.now()) {
    return console.log(`[server]    ${peer.address} 无权限，丢弃入站包`)
  }
  const ch = session.peerChannels.get(`${peer.address}:${peer.port}`)
  if (ch !== undefined) {
    // 已绑定通道：只加 4 字节 ChannelData 头
    const head = Buffer.alloc(4)
    head.writeUInt16BE(ch, 0)
    head.writeUInt16BE(data.length, 2)
    server.send(Buffer.concat([head, data, Buffer.alloc(pad4(data.length) - data.length)]),
      session.client.port, session.client.address)
    console.log(`[server]    ChannelData(ch=0x${ch.toString(16)})：${peer.address}:${peer.port} -> client，${data.length}B`)
  } else {
    // 未绑定：用 Data indication 包装
    const ind = buildMessage(encodeType(M.Data, CLS.Indication), crypto.randomBytes(12).toString('hex'), [
      { type: A.XorPeerAddress, value: encodeXorAddress(peer.address, peer.port) },
      { type: A.Data, value: data },
    ])
    server.send(ind, session.client.port, session.client.address)
    console.log(`[server]    Data indication：${peer.address}:${peer.port} -> client，${data.length}B`)
  }
}

// ---------- ChannelBind ----------
function onChannelBind(m, rinfo) {
  const ckey = clientKeyOf(rinfo)
  const session = sessions.get(ckey)
  console.log(`[server] <- ChannelBind（来自 ${ckey}）`)
  if (!session) return sendError(rinfo, m.txnId, M.ChannelBind, 437, 'Allocation Mismatch', [], null)
  if (!verifyMessageIntegrity(m.raw, session.key)) {
    return sendError(rinfo, m.txnId, M.ChannelBind, 401, 'Unauthorized', [
      { type: A.Realm, value: Buffer.from(REALM) },
      { type: A.Nonce, value: Buffer.from(nonces.get(ckey)) },
    ])
  }
  const ch = getAttr(m.attrs, A.ChannelNumber).readUInt16BE(0)
  const peer = decodeXorAddress(getAttr(m.attrs, A.XorPeerAddress))
  if (ch < 0x4000 || ch > 0x7fff) {
    return sendError(rinfo, m.txnId, M.ChannelBind, 400, 'Bad Request', [], session.key)
  }
  session.channels.set(ch, peer)
  session.peerChannels.set(`${peer.addr}:${peer.port}`, ch)
  const ok = buildMessage(encodeType(M.ChannelBind, CLS.Success), m.txnId, [])
  reply(rinfo, signMessage(ok, session.key))
  console.log(`[server] -> ChannelBind success，通道 0x${ch.toString(16)} <-> ${peer.addr}:${peer.port}`)
}

// ---------- ChannelData（client -> peer，4 字节头）----------
function onChannelData(msg, rinfo) {
  const session = sessions.get(clientKeyOf(rinfo))
  if (!session) return
  const ch = msg.readUInt16BE(0)
  const len = msg.readUInt16BE(2)
  const data = msg.subarray(4, 4 + len)
  const peer = session.channels.get(ch)
  if (!peer) return
  session.relay.send(data, peer.port, peer.addr)
  console.log(`[server]    ChannelData(ch=0x${ch.toString(16)})：client -> ${peer.addr}:${peer.port}，${data.length}B`)
}

// ---------- Refresh（续期 / 释放）----------
function onRefresh(m, rinfo) {
  const ckey = clientKeyOf(rinfo)
  const session = sessions.get(ckey)
  console.log(`[server] <- Refresh（来自 ${ckey}）`)
  if (!session) return sendError(rinfo, m.txnId, M.Refresh, 437, 'Allocation Mismatch', [], null)
  if (!verifyMessageIntegrity(m.raw, session.key)) {
    return sendError(rinfo, m.txnId, M.Refresh, 401, 'Unauthorized', [
      { type: A.Realm, value: Buffer.from(REALM) },
      { type: A.Nonce, value: Buffer.from(nonces.get(ckey)) },
    ])
  }
  const ltBuf = getAttr(m.attrs, A.Lifetime)
  const lifetime = ltBuf ? ltBuf.readUInt32BE(0) : ALLOCATION_LIFETIME
  if (lifetime === 0) {
    session.relay.close()
    sessions.delete(ckey)
    console.log('[server]    租约归零，分配已释放')
  } else {
    session.expireAt = Date.now() + lifetime * 1000
  }
  const ok = buildMessage(encodeType(M.Refresh, CLS.Success), m.txnId, [
    { type: A.Lifetime, value: u32(lifetime) },
  ])
  reply(rinfo, signMessage(ok, session.key))
  console.log(`[server] -> Refresh success，LIFETIME=${lifetime}`)
}

server.bind(PORT, HOST, () => {
  console.log(`[server] mini TURN 服务器监听于 ${HOST}:${PORT}（REALM=${REALM}，用户=${Object.keys(USERS).join(',')}）`)
})

12.3.3 turn-client.js ​

保存为 turn-client.js（编解码工具与服务器相同，为便于单文件运行而复制一份）：

js
/**
 * turn-client.js
 * 手写 mini TURN 客户端：演示完整事务流程
 */
'use strict'
const dgram = require('dgram')
const crypto = require('crypto')

// ==================== 协议常量（与服务器一致）====================
const SERVER = { host: '127.0.0.1', port: 3478 }
const USERNAME = 'webrtc'
const PASSWORD = 'webrtc2024'
const CHANNEL = 0x4001          // 通道号必须在 0x4000-0x7FFF
const MAGIC_COOKIE = 0x2112a442

const M = { Allocate: 0x003, Refresh: 0x004, Send: 0x006, Data: 0x007, CreatePermission: 0x008, ChannelBind: 0x009 }
const CLS = { Request: 0, Indication: 1, Success: 2, Error: 3 }
const A = {
  Username: 0x0006, MessageIntegrity: 0x0008, ErrorCode: 0x0009,
  ChannelNumber: 0x000c, Lifetime: 0x000d, XorPeerAddress: 0x0012,
  Data: 0x0013, Realm: 0x0014, Nonce: 0x0015, XorRelayedAddress: 0x0016,
  RequestedTransport: 0x0019, XorMappedAddress: 0x0020,
}

// ==================== 编解码（与服务器相同）====================
function encodeType(method, klass) {
  return (method & 0x000f) | ((method & 0x0070) << 1) |
    ((method & 0x0f80) << 2) | ((klass & 0x01) << 4) | ((klass & 0x02) << 7)
}
function decodeType(t) {
  return {
    method: (t & 0x000f) | ((t & 0x00e0) >> 1) | ((t & 0x3e00) >> 2),
    klass: ((t & 0x0010) >> 4) | ((t & 0x0100) >> 7),
  }
}
const pad4 = (n) => (n + 3) & ~3

function buildMessage(type, txnIdHex, attrs) {
  let body = Buffer.alloc(0)
  for (const a of attrs) {
    const head = Buffer.alloc(4)
    head.writeUInt16BE(a.type, 0)
    head.writeUInt16BE(a.value.length, 2)
    body = Buffer.concat([body, head, a.value, Buffer.alloc(pad4(a.value.length) - a.value.length)])
  }
  const head = Buffer.alloc(20)
  head.writeUInt16BE(type, 0)
  head.writeUInt16BE(body.length, 2)
  head.writeUInt32BE(MAGIC_COOKIE, 4)
  Buffer.from(txnIdHex, 'hex').copy(head, 8)
  return Buffer.concat([head, body])
}
function parseMessage(msg) {
  const m = decodeType(msg.readUInt16BE(0))
  const len = msg.readUInt16BE(2)
  m.txnId = msg.subarray(8, 20).toString('hex')
  m.attrs = []
  let off = 20
  while (off + 4 <= 20 + len) {
    const t = msg.readUInt16BE(off)
    const l = msg.readUInt16BE(off + 2)
    m.attrs.push({ type: t, value: msg.subarray(off + 4, off + 4 + l) })
    off += 4 + pad4(l)
  }
  return m
}
const getAttr = (attrs, type) => {
  const a = attrs.find((x) => x.type === type)
  return a ? a.value : null
}
function encodeXorAddress(addr, port) {
  const buf = Buffer.alloc(8)
  buf[1] = 0x01
  buf.writeUInt16BE(port ^ 0x2112, 2)
  const ip = addr.split('.').map(Number)
  const ck = [0x21, 0x12, 0xa4, 0x42]
  for (let i = 0; i < 4; i++) buf[4 + i] = ip[i] ^ ck[i]
  return buf
}
function decodeXorAddress(buf) {
  const ck = [0x21, 0x12, 0xa4, 0x42]
  return {
    port: buf.readUInt16BE(2) ^ 0x2112,
    addr: [0, 1, 2, 3].map((i) => buf[4 + i] ^ ck[i]).join('.'),
  }
}
const u32 = (n) => { const b = Buffer.alloc(4); b.writeUInt32BE(n, 0); return b }
const longTermKey = (u, r, p) => crypto.createHash('md5').update(`${u}:${r}:${p}`).digest()
function signMessage(msg, key) {
  msg.writeUInt16BE(msg.readUInt16BE(2) + 24, 2)
  const mac = crypto.createHmac('sha1', key).update(msg).digest()
  const mi = Buffer.alloc(24)
  mi.writeUInt16BE(A.MessageIntegrity, 0)
  mi.writeUInt16BE(20, 2)
  mac.copy(mi, 4)
  return Buffer.concat([msg, mi])
}

// ==================== 客户端主体 ====================
const client = dgram.createSocket('udp4')
const pending = new Map()   // txnId -> resolve（匹配响应）
const waiters = []          // 等待 indication / ChannelData

const sleep = (ms) => new Promise((r) => setTimeout(r, ms))
const once = (type) => new Promise((resolve) => waiters.push({ type, resolve }))
const fire = (type, payload) => {
  const i = waiters.findIndex((w) => w.type === type)
  if (i >= 0) waiters.splice(i, 1)[0].resolve(payload)
}
const newTxn = () => crypto.randomBytes(12).toString('hex')

function sendAndWait(buf) {
  const txnId = buf.subarray(8, 20).toString('hex')
  return new Promise((resolve) => {
    pending.set(txnId, resolve)
    client.send(buf, SERVER.port, SERVER.host)
  })
}

client.on('message', (msg) => {
  const first16 = msg.readUInt16BE(0)
  // ChannelData：高两位 01
  if (first16 >= 0x4000 && first16 < 0x8000) {
    const len = msg.readUInt16BE(2)
    const data = msg.subarray(4, 4 + len)
    console.log(`[client] <- ChannelData(ch=0x${first16.toString(16)})，${len}B："${data.toString()}"`)
    return fire('channeldata', { ch: first16, data })
  }
  const m = parseMessage(msg)
  if (pending.has(m.txnId)) {
    pending.get(m.txnId)(m)
    pending.delete(m.txnId)
  } else if (m.method === M.Data && m.klass === CLS.Indication) {
    const peer = decodeXorAddress(getAttr(m.attrs, A.XorPeerAddress))
    const data = getAttr(m.attrs, A.Data)
    console.log(`[client] <- Data indication，来自 ${peer.addr}:${peer.port}，${data.length}B："${data.toString()}"`)
    fire('dataind', { peer, data })
  }
})

async function main() {
  await new Promise((r) => client.bind(0, '127.0.0.1', r))
  console.log(`[client] 客户端 UDP ${client.address().address}:${client.address().port}`)

  // ---- 1. 第一次 Allocate：不带凭证，期待 401 ----
  console.log('\n[client] -> Allocate（无凭证）')
  let res = await sendAndWait(buildMessage(encodeType(M.Allocate, CLS.Request), newTxn(), [
    { type: A.RequestedTransport, value: Buffer.from([17, 0, 0, 0]) }, // 17 = UDP
  ]))
  const err = getAttr(res.attrs, A.ErrorCode)
  console.log(`[client] <- Allocate error：code=${err[2]}${String(err[3]).padStart(2, '0')}（预期 401）`)
  const realm = getAttr(res.attrs, A.Realm).toString()
  const nonce = getAttr(res.attrs, A.Nonce).toString()
  console.log(`[client]    REALM=${realm}，NONCE=${nonce}`)

  // ---- 2. 第二次 Allocate：带长期凭证 + MESSAGE-INTEGRITY ----
  const key = longTermKey(USERNAME, realm, PASSWORD)
  console.log('\n[client] -> Allocate（带 USERNAME/REALM/NONCE + MESSAGE-INTEGRITY）')
  const authAttrs = () => [
    { type: A.Username, value: Buffer.from(USERNAME) },
    { type: A.Realm, value: Buffer.from(realm) },
    { type: A.Nonce, value: Buffer.from(nonce) },
  ]
  res = await sendAndWait(signMessage(buildMessage(
    encodeType(M.Allocate, CLS.Request), newTxn(),
    [...authAttrs(), { type: A.RequestedTransport, value: Buffer.from([17, 0, 0, 0]) }],
  ), key))
  const relayed = decodeXorAddress(getAttr(res.attrs, A.XorRelayedAddress))
  const lifetime = getAttr(res.attrs, A.Lifetime).readUInt32BE(0)
  console.log(`[client] <- Allocate success！relayed address = ${relayed.addr}:${relayed.port}，租约 ${lifetime}s`)

  // ---- 3. 模拟对端 peer ----
  const peer = dgram.createSocket('udp4')
  await new Promise((r) => peer.bind(0, '127.0.0.1', r))
  const peerAddr = peer.address()
  console.log(`[peer]   对端 UDP ${peerAddr.address}:${peerAddr.port}`)
  peer.on('message', (data, from) => {
    console.log(`[peer]   <- 收到 ${from.address}:${from.port} 的数据："${data.toString()}"`)
  })

  // ---- 4. CreatePermission：允许 peer 的 IP ----
  console.log('\n[client] -> CreatePermission（peer = 127.0.0.1）')
  res = await sendAndWait(signMessage(buildMessage(
    encodeType(M.CreatePermission, CLS.Request), newTxn(),
    [...authAttrs(), { type: A.XorPeerAddress, value: encodeXorAddress('127.0.0.1', peerAddr.port) }],
  ), key))
  console.log('[client] <- CreatePermission success')

  // ---- 5. peer 发到 relayed address，client 收 Data indication ----
  console.log('\n[peer]   -> 发数据到 relayed address（走 Data indication 通路）')
  const waitInd = once('dataind')
  peer.send('你好，TURN 中继！', relayed.port, relayed.addr)
  await waitInd

  // ---- 6. ChannelBind：绑定通道 0x4001 <-> peer ----
  console.log('\n[client] -> ChannelBind（ch=0x4001 <-> peer）')
  const chBuf = Buffer.alloc(4)
  chBuf.writeUInt16BE(CHANNEL, 0)
  res = await sendAndWait(signMessage(buildMessage(
    encodeType(M.ChannelBind, CLS.Request), newTxn(),
    [...authAttrs(),
      { type: A.ChannelNumber, value: chBuf },
      { type: A.XorPeerAddress, value: encodeXorAddress('127.0.0.1', peerAddr.port) }],
  ), key))
  console.log('[client] <- ChannelBind success，通道 0x4001 已绑定')

  // ---- 7. peer 再发一次：这次走 ChannelData ----
  console.log('\n[peer]   -> 再发一次（走 ChannelData 通路）')
  const waitCh = once('channeldata')
  peer.send('这条走 ChannelData', relayed.port, relayed.addr)
  await waitCh

  // ---- 8. client -> peer：Send indication ----
  console.log('\n[client] -> Send indication（client -> peer）')
  client.send(buildMessage(encodeType(M.Send, CLS.Indication), newTxn(), [
    { type: A.XorPeerAddress, value: encodeXorAddress('127.0.0.1', peerAddr.port) },
    { type: A.Data, value: Buffer.from('客户端经 Send indication 回话') },
  ]), SERVER.port, SERVER.host)
  await sleep(300)

  // ---- 9. client -> peer：ChannelData ----
  console.log('\n[client] -> ChannelData（client -> peer）')
  const payload = Buffer.from('客户端经 ChannelData 回话')
  const chd = Buffer.alloc(4)
  chd.writeUInt16BE(CHANNEL, 0)
  chd.writeUInt16BE(payload.length, 2)
  client.send(Buffer.concat([chd, payload, Buffer.alloc(pad4(payload.length) - payload.length)]),
    SERVER.port, SERVER.host)
  await sleep(300)

  // ---- 10. Refresh LIFETIME=0：主动释放分配 ----
  console.log('\n[client] -> Refresh（LIFETIME=0，释放分配）')
  res = await sendAndWait(signMessage(buildMessage(
    encodeType(M.Refresh, CLS.Request), newTxn(),
    [...authAttrs(), { type: A.Lifetime, value: u32(0) }],
  ), key))
  console.log(`[client] <- Refresh success，LIFETIME=${getAttr(res.attrs, A.Lifetime).readUInt32BE(0)}，分配已释放`)

  client.close()
  peer.close()
  console.log('\n[client] 全流程完成 ✅')
}

main().catch((e) => { console.error(e); process.exit(1) })

12.3.5 与生产级 Coturn 的差距 ​

mini 版是教学实现，要上到生产还差这些：

能力mini 版Coturn（生产）
传输仅 UDPUDP / TCP / TLS / DTLS
地址族仅 IPv4IPv4 / IPv6 双栈
凭证存储硬编码单用户数据库 / Redis / REST API 短期凭证
资源回收无定时器（进程退出才释放）精确的 600s/300s 到期回收
并发未优化单机 5000~10000 并发
FINGERPRINT 属性未实现支持（防协议伪装检测）
公平带宽分配无每分配限速

手写完这个 mini 版，再回去读 RFC 5766 或 Coturn 的配置项，你会发现每个名词都对得上号了。
